29// The Verilog only version does not match regular expressions,
30// it only does glob style matching.
31function int uvm_re_match(string re, string str);
32 int e, es, s, ss;
33 string tmp;
34 e = 0; s = 0;
35 es = 0; ss = 0;
36 
37 if(re.len() == 0)
38 return 0;
39 
40 // The ^ used to be used to remove the implicit wildcard, but now we don't
41 // use implicit wildcard so this character is just stripped.
42 if(re[0] == "^")
43 re = re.substr(1, re.len()-1);
44 
45 //This loop is only needed when the first character of the re may not
46 //be a *.
47 while (s != str.len() && re.getc(e) != "*") begin
48 if ((re.getc(e) != str.getc(s)) && (re.getc(e) != "?"))
49 return 1;
50 e++; s++;
51 end
52 
53 while (s != str.len()) begin
54 if (re.getc(e) == "*") begin
55 e++;
56 if (e == re.len()) begin
57 return 0;
58 end
59 es = e;
60 ss = s+1;
61 end
62 else if (re.getc(e) == str.getc(s) || re.getc(e) == "?") begin
63 e++;
64 s++;
65 end
66 else begin
67 e = es;
68 s = ss++;
69 end
70 end
71 while (e < re.len() && re.getc(e) == "*")
72 e++;
73 if(e == re.len()) begin
74 return 0;
75 end
76 else begin
77 return 1;
78 end
79endfunction
