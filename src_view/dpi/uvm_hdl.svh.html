77 // Function: uvm_hdl_force_time
78 //
79 // Forces the ~value~ on the given ~path~ for the specified amount of ~force_time~.
80 // If ~force_time~ is 0, <uvm_hdl_deposit> is called.
81 // Returns 1 if the call succeeded, 0 otherwise.
82 //
83 task uvm_hdl_force_time(string path, uvm_hdl_data_t value, time force_time = 0);
84 if (force_time == 0) begin
85 void'(uvm_hdl_deposit(path, value));
86 return;
87 end
88 if (!uvm_hdl_force(path, value))
89 return;
90 #force_time;
91 void'(uvm_hdl_release_and_read(path, value));
92 endtask

95 // Function: uvm_hdl_release_and_read
96 //
97 // Releases a value previously set with <uvm_hdl_force>.
98 // Returns 1 if the call succeeded, 0 otherwise. ~value~ is set to
99 // the HDL value after the release. For 'reg', the value will still be
100 // the forced value until it has been procedurally reassigned. For 'wire',
101 // the value will change immediately to the resolved value of its
102 // continuous drivers, if any. If none, its value remains as forced until
103 // the next direct assignment.
104 //
105 import "DPI-C" context function int uvm_hdl_release_and_read(string path, inout uvm_hdl_data_t value);
