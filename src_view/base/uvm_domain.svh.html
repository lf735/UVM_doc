190 // Function: jump
191 //
192 // jumps all active phases of this domain to to-phase if
193 // there is a path between active-phase and to-phase
194 function void jump(uvm_phase phase);
195 uvm_phase phases[$];
196 
197 m_get_transitive_children(phases);
198
199 phases = phases.find(item) with (item.get_state() inside {[UVM_PHASE_STARTED:UVM_PHASE_CLEANUP]});
200
201 foreach(phases[idx])
202 if(phases[idx].is_before(phase) || phases[idx].is_after(phase))
203 phases[idx].jump(phase);
204
205 endfunction
