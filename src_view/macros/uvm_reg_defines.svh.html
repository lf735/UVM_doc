25// Macro: `UVM_REG_ADDR_WIDTH

47// Macro: `UVM_REG_BYTENABLE_WIDTH
48//
49// Maximum number of byte enable bits
50//
51// Default value is one per byte in <`UVM_REG_DATA_WIDTH>.
52// Used to define the <uvm_reg_byte_en_t> type.
53//
54`ifndef UVM_REG_BYTENABLE_WIDTH
55 `define UVM_REG_BYTENABLE_WIDTH ((`UVM_REG_DATA_WIDTH-1)/8+1)
56`endif
