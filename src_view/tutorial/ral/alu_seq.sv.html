19// Pipelined response-aware leaf sequence using grab()/lock() arbitration
20class alu_smoke_seq extends uvm_sequence #(alu_item);
21 `uvm_object_utils(alu_smoke_seq)
22 
23 rand int unsigned num_items = 6;
24 rand bit use_lock = 1'b0;
25 
26 function new(string name = "alu_smoke_seq");
27 super.new(name);
28 endfunction
29 
30 virtual task body();
31 alu_item req, rsp;
32 if (use_lock) m_sequencer.lock(this);
33 
34 for (int i = 0; i < num_items; i++) begin
35 req = alu_item::type_id::create($sformatf("req_%0d", i));
36 start_item(req);
37 if (!req.randomize() with { tag == local::i[3:0]; }) begin
38 `uvm_fatal("SEQ", "Randomization failed")
39 end
40 finish_item(req);
41 get_response(rsp);
42 `uvm_info("SEQ", $sformatf("Received rsp: %s", rsp.convert2string()), UVM_HIGH)
43 end
44 
45 if (use_lock) m_sequencer.unlock(this);
46 endtask
47endclass

49// Layered Translator Sequence: converts high-level frame "(x + y) * scale"
50// into two consecutive low-level alu_item operations on the data sequencer.
51class alu_layered_frame_seq extends uvm_sequence #(alu_item);
52 `uvm_object_utils(alu_layered_frame_seq)
53 
54 alu_macro_frame_item frame;
55 
56 function new(string name = "alu_layered_frame_seq");
57 super.new(name);
58 endfunction
59 
60 virtual task body();
61 alu_item step1, step2, rsp1, rsp2;
62 if (frame == null) begin
63 frame = alu_macro_frame_item::type_id::create("frame");
64 void'(frame.randomize() with { x inside {[1:50]}; y inside {[1:50]}; scale inside {[2:4]}; });
65 end
66 
67 // Layer Step 1: sum = x + y
68 step1 = alu_item::type_id::create("step1_add");
69 start_item(step1);
70 step1.op = ALU_ADD;
71 step1.a = frame.x;
72 step1.b = frame.y;
73 step1.tag = 4'hA;
74 finish_item(step1);
75 get_response(rsp1);
76 
77 // Layer Step 2: result = sum * scale (feeds output of Step 1 into Step 2!)
78 step2 = alu_item::type_id::create("step2_mul");
79 start_item(step2);
80 step2.op = ALU_MUL;
81 step2.a = rsp1.result[15:0];
82 step2.b = frame.scale;
83 step2.tag = 4'hB;
84 finish_item(step2);
85 get_response(rsp2);
86 
87 `uvm_info("LAYERED_SEQ", $sformatf("Macro (%0d + %0d) * %0d = %0d",
88 frame.x, frame.y, frame.scale, rsp2.result), UVM_LOW)
89 endtask
90endclass
