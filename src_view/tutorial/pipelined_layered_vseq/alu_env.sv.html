30 data_agent = alu_agent::type_id::create("data_agent", this);

32 v_seqr = alu_virtual_sequencer::type_id::create("v_seqr", this);
33 
34 if (cfg.enable_scoreboard) scb = alu_scoreboard::type_id::create("scb", this);

40 // Bind sub-sequencer handles onto the virtual sequencer
41 if (data_agent.seqr != null) v_seqr.alu_sqr = data_agent.seqr;
42 if (apb_agt.seqr != null) v_seqr.apb_sqr = apb_agt.seqr;
43 
44 if (cfg.enable_scoreboard) begin
45 data_agent.req_ap.connect(scb.req_export);

47 end

49 data_agent.req_ap.connect(cov.analysis_export);
50 end
