7class apb_item extends uvm_sequence_item;
8 rand bit pwrite;
9 rand logic [7:0] paddr;
10 rand logic [31:0] pwdata;
11 logic [31:0] prdata;

13 `uvm_object_utils_begin(apb_item)
14 `uvm_field_int(pwrite, UVM_ALL_ON)
15 `uvm_field_int(paddr, UVM_ALL_ON | UVM_HEX)
16 `uvm_field_int(pwdata, UVM_ALL_ON | UVM_HEX)
17 `uvm_field_int(prdata, UVM_ALL_ON | UVM_HEX)
18 `uvm_object_utils_end

39class apb_driver extends uvm_driver #(apb_item);
40 `uvm_component_utils(apb_driver)
41 apb_agent_cfg cfg;
42 virtual apb_if vif;
43 
44 function new(string name = "apb_driver", uvm_component parent = null);
45 super.new(name, parent);
46 endfunction
47 
48 virtual function void build_phase(uvm_phase phase);
49 super.build_phase(phase);
50 if (!uvm_config_db#(apb_agent_cfg)::get(this, "", "cfg", cfg))
51 `uvm_fatal("APB_DRV", "Missing apb_agent_cfg")
52 vif = cfg.vif;
53 endfunction
54 
55 virtual task run_phase(uvm_phase phase);
56 apb_item req;
57 vif.reset_signals();
58 @(posedge vif.rst_n);
59 @(vif.drv_cb);
60 forever begin
61 seq_item_port.get_next_item(req);
62 // SETUP phase
63 vif.drv_cb.psel <= 1'b1;
64 vif.drv_cb.penable <= 1'b0;
65 vif.drv_cb.pwrite <= req.pwrite;
66 vif.drv_cb.paddr <= req.paddr;
67 vif.drv_cb.pwdata <= req.pwdata;
68 @(vif.drv_cb);
69 // ACCESS phase
70 vif.drv_cb.penable <= 1'b1;
71 do @(vif.drv_cb); while (!vif.drv_cb.pready);
72 if (!req.pwrite) req.prdata = vif.drv_cb.prdata;
73 vif.drv_cb.psel <= 1'b0;
74 vif.drv_cb.penable <= 1'b0;
75 seq_item_port.item_done();
76 end
77 endtask
78endclass

89 
90 virtual function void build_phase(uvm_phase phase);
91 super.build_phase(phase);
92 ap = new("ap", this);
93 if (!uvm_config_db#(apb_agent_cfg)::get(this, "", "cfg", cfg))
94 `uvm_fatal("APB_MON", "Missing apb_agent_cfg")
95 vif = cfg.vif;
96 endfunction
97 
98 virtual task run_phase(uvm_phase phase);
99 @(posedge vif.rst_n);
100 forever begin
101 @(vif.mon_cb);
102 if (vif.mon_cb.psel && vif.mon_cb.penable && vif.mon_cb.pready) begin
103 apb_item tr = apb_item::type_id::create("apb_tr");
104 tr.pwrite = vif.mon_cb.pwrite;
105 tr.paddr = vif.mon_cb.paddr;
106 tr.pwdata = vif.mon_cb.pwdata;
107 tr.prdata = vif.mon_cb.prdata;
108 ap.write(tr);
109 end
110 end
111 endtask
112endclass

114class apb_agent extends uvm_agent;
115 `uvm_component_utils(apb_agent)
116 apb_agent_cfg cfg;

118 apb_driver drv;

122 function new(string name = "apb_agent", uvm_component parent = null);
123 super.new(name, parent);
124 endfunction
125 
126 virtual function void build_phase(uvm_phase phase);
127 super.build_phase(phase);
128 ap = new("ap", this);
129 if (!uvm_config_db#(apb_agent_cfg)::get(this, "", "cfg", cfg))
130 `uvm_fatal("APB_AGT", "Missing apb_agent_cfg")
131 uvm_config_db#(apb_agent_cfg)::set(this, "mon", "cfg", cfg);
132 mon = apb_monitor::type_id::create("mon", this);
133 if (cfg.is_active == UVM_ACTIVE) begin
134 uvm_config_db#(apb_agent_cfg)::set(this, "drv", "cfg", cfg);
135 seqr = uvm_sequencer#(apb_item)::type_id::create("seqr", this);
136 drv = apb_driver::type_id::create("drv", this);
137 end
138 endfunction
139 
140 virtual function void connect_phase(uvm_phase phase);
141 super.connect_phase(phase);
142 mon.ap.connect(ap);
143 if (cfg.is_active == UVM_ACTIVE) drv.seq_item_port.connect(seqr.seq_item_export);
144 endfunction
145endclass
