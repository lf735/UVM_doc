3// Registers:
4// 0x00 CTRL : [0] EN, [1] SAT_EN, [2] ERR_MASK

8// 0x10 ERR_INJECT : [0] FORCE_ERR (diagnostic register)

45 // Register storage (accessible via RAL backdoor HDL paths)
46 logic [31:0] reg_ctrl;
47 logic [31:0] reg_status;
48 logic [31:0] reg_sat_limit;

51 
52 wire cfg_en = reg_ctrl[0];
53 wire cfg_sat_en = reg_ctrl[1];
54 wire cfg_err_mask = reg_ctrl[2];
55 wire force_err = reg_err_inject[0];
56 
57 assign in_ready = cfg_en && (!out_valid || out_ready);
58 assign pready = 1'b1; // Zero-wait-state APB slave

60 // APB3 Register Read/Write Logic
61 always_ff @(posedge clk or negedge rst_n) begin
62 if (!rst_n) begin
63 reg_ctrl <= 32'h0000_0001; // EN=1 by default
64 reg_status <= 32'h0000_0000;
65 reg_sat_limit <= 32'h0000_FFFF;
66 reg_pkt_count <= 32'h0000_0000;
67 reg_err_inject <= 32'h0000_0000;
68 end else begin
69 // Update BUSY bit
70 reg_status[0] <= out_valid;
71 
72 if (psel && penable && pwrite) begin
73 case (paddr)
74 8'h00: reg_ctrl <= pwdata & 32'h0000_0007;
75 8'h04: reg_status[1] <= reg_status[1] & ~pwdata[1]; // W1C LAST_ERR
76 8'h08: reg_sat_limit <= pwdata & 32'h0000_FFFF;
77 8'h10: reg_err_inject <= pwdata & 32'h0000_0001;
78 default: ;
79 endcase
80 end
81 
82 if (in_valid && in_ready) begin
83 reg_pkt_count <= reg_pkt_count + 32'd1;
84 if (force_err || (op > OP_PASS_A)) begin
85 reg_status[1] <= 1'b1;
86 end
87 end
88 end
89 end
90 
91 // Combinational APB read mux
92 always_comb begin
93 prdata = 32'h0000_0000;
94 if (psel && !pwrite) begin
95 case (paddr)
96 8'h00: prdata = reg_ctrl;
97 8'h04: prdata = reg_status;
98 8'h08: prdata = reg_sat_limit;
99 8'h0C: prdata = reg_pkt_count;
100 8'h10: prdata = reg_err_inject;
101 default: prdata = 32'h0000_0000;
102 endcase
103 end
104 end

106 // Packet-ALU Datapath with Saturation
107 logic [31:0] raw_res;
108 logic raw_err;
109 
110 always_comb begin
111 raw_err = force_err;
112 case (op)
113 OP_ADD: raw_res = {16'd0, a} + {16'd0, b};
114 OP_SUB: raw_res = {16'd0, a} - {16'd0, b};
115 OP_MUL: raw_res = a * b;
116 OP_XOR: raw_res = {16'd0, (a ^ b)};
117 OP_PASS_A: raw_res = {16'd0, a};
118 default: begin
119 raw_res = 32'hDEAD_BEEF;
120 raw_err = 1'b1;
121 end
122 endcase
123 
124 if (!raw_err && cfg_sat_en && (raw_res > {16'd0, reg_sat_limit[15:0]})) begin
125 raw_res = {16'd0, reg_sat_limit[15:0]};
126 end
127 end
128 
129 always_ff @(posedge clk or negedge rst_n) begin
130 if (!rst_n) begin
131 out_valid <= 1'b0;
132 result <= 32'd0;
133 out_tag <= 4'd0;
134 err_flag <= 1'b0;
135 end else begin
136 if (in_valid && in_ready) begin
137 out_valid <= 1'b1;
138 out_tag <= tag;
139 result <= raw_res;
140 err_flag <= raw_err & ~cfg_err_mask;
141 end else if (out_valid && out_ready) begin
142 out_valid <= 1'b0;
143 end
144 end
145 end
