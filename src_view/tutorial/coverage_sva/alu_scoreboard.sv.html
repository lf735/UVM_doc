4`ifndef ALU_SCOREBOARD_SV
5`define ALU_SCOREBOARD_SV
6 
7class alu_scoreboard extends uvm_scoreboard;
8 `uvm_component_utils(alu_scoreboard)
9 
10 uvm_analysis_export #(alu_item) req_export;
11 uvm_analysis_export #(alu_item) rsp_export;
12 uvm_tlm_analysis_fifo #(alu_item) req_fifo;
13 uvm_tlm_analysis_fifo #(alu_item) rsp_fifo;
14 
15 alu_ref_model ref_model;
16 int unsigned pass_count;
17 int unsigned fail_count;
18 
19 function new(string name = "alu_scoreboard", uvm_component parent = null);
20 super.new(name, parent);
21 endfunction
22 
23 virtual function void build_phase(uvm_phase phase);
24 super.build_phase(phase);
25 req_export = new("req_export", this);
26 rsp_export = new("rsp_export", this);
27 req_fifo = new("req_fifo", this);
28 rsp_fifo = new("rsp_fifo", this);
29 ref_model = alu_ref_model::type_id::create("ref_model", this);
30 endfunction
31 
32 virtual function void connect_phase(uvm_phase phase);
33 super.connect_phase(phase);
34 req_export.connect(req_fifo.analysis_export);
35 rsp_export.connect(rsp_fifo.analysis_export);
36 endfunction
37 
38 virtual task run_phase(uvm_phase phase);
39 alu_item req, exp, act;
40 forever begin
41 req_fifo.get(req);
42 rsp_fifo.get(act);
43 exp = ref_model.predict(req);
44 
45 if (act.tag !== exp.tag || act.result !== exp.result || act.err_flag !== exp.err_flag) begin
46 fail_count++;
47 `uvm_error("SCB", $sformatf(
48 "MISMATCH! EXP(tag=%0d res=0x%08h err=%0b) != ACT(tag=%0d res=0x%08h err=%0b)",
49 exp.tag, exp.result, exp.err_flag, act.tag, act.result, act.err_flag))
50 end else begin
51 pass_count++;
52 `uvm_info("SCB", $sformatf("MATCH tag=%0d op=%s result=0x%08h",
53 exp.tag, exp.op.name(), act.result), UVM_MEDIUM)
54 end
55 end
56 endtask
57 
58 virtual function void check_phase(uvm_phase phase);
59 super.check_phase(phase);
60 if (!req_fifo.is_empty() || !rsp_fifo.is_empty()) begin
61 `uvm_error("SCB", $sformatf("Unmatched items at end of test: req=%0d rsp=%0d",
62 req_fifo.used(), rsp_fifo.used()))
63 end
64 endfunction
65 
66 virtual function void report_phase(uvm_phase phase);
67 super.report_phase(phase);
68 `uvm_info("SCB", $sformatf("Scoreboard Summary: PASS=%0d FAIL=%0d", pass_count, fail_count), UVM_LOW)
69 endfunction
70endclass
71 
72`endif
