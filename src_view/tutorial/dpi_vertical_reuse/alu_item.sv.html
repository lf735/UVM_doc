4`ifndef ALU_ITEM_SV
5`define ALU_ITEM_SV
6 
7typedef enum logic [2:0] {
8 ALU_ADD = 3'd0,
9 ALU_SUB = 3'd1,
10 ALU_MUL = 3'd2,
11 ALU_XOR = 3'd3,
12 ALU_PASS_A = 3'd4,
13 ALU_INVAL = 3'd5
14} alu_op_e;
15 
16class alu_item extends uvm_sequence_item;
17 rand alu_op_e op;
18 rand logic [15:0] a;
19 rand logic [15:0] b;
20 rand logic [3:0] tag;
21 
22 logic [31:0] result;
23 logic err_flag;
24 
25 constraint c_valid_op {
26 op dist { [ALU_ADD:ALU_PASS_A] := 9, ALU_INVAL := 1 };
27 }
28 
29 constraint c_operand_range {
30 a dist { 16'h0000 := 1, [16'h0001:16'hFFFE] := 8, 16'hFFFF := 1 };
31 b inside {[16'h0000 : 16'hFFFF]};
32 }
33 
34 `uvm_object_utils_begin(alu_item)
35 `uvm_field_enum(alu_op_e, op, UVM_ALL_ON)
36 `uvm_field_int(a, UVM_ALL_ON | UVM_HEX)
37 `uvm_field_int(b, UVM_ALL_ON | UVM_HEX)
38 `uvm_field_int(tag, UVM_ALL_ON | UVM_DEC)
39 `uvm_field_int(result, UVM_ALL_ON | UVM_HEX)
40 `uvm_field_int(err_flag, UVM_ALL_ON)
41 `uvm_object_utils_end
42 
43 function new(string name = "alu_item");
44 super.new(name);
45 endfunction

47 virtual function string convert2string();
48 return $sformatf("tag=%0d op=%s a=0x%04h b=0x%04h res=0x%08h err=%0b",
