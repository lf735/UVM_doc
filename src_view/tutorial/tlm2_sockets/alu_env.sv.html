7class alu_env extends uvm_env;

9 
10 alu_env_cfg cfg;
11 alu_agent data_agent;
12 apb_agent apb_agt;
13 alu_virtual_sequencer v_seqr;
14 alu_tlm2_bridge tlm2_bridge;
15 alu_scoreboard scb;
16 alu_cov_subscriber cov;
17 
18 pkt_alu_reg_block reg_model;
19 apb_reg_adapter reg_adapter;
20 uvm_reg_predictor #(apb_item) reg_predictor;
21 
22 uvm_domain apb_domain;
23 uvm_heartbeat hb;
24 uvm_objection hb_obj;

27 function new(string name = "alu_env", uvm_component parent = null);
28 super.new(name, parent);
29 endfunction
30 
31 virtual function void build_phase(uvm_phase phase);
32 super.build_phase(phase);
33 if (!uvm_config_db#(alu_env_cfg)::get(this, "", "env_cfg", cfg))
34 `uvm_fatal("ENV", "alu_env_cfg not found")
35 
36 uvm_config_db#(alu_agent_cfg)::set(this, "data_agent", "cfg", cfg.data_agent_cfg);
37 uvm_config_db#(apb_agent_cfg)::set(this, "apb_agt", "cfg", cfg.apb_cfg);
38 
39 data_agent = alu_agent::type_id::create("data_agent", this);
40 apb_agt = apb_agent::type_id::create("apb_agt", this);
41 v_seqr = alu_virtual_sequencer::type_id::create("v_seqr", this);
42 tlm2_bridge = alu_tlm2_bridge::type_id::create("tlm2_bridge", this);
43 reg_model = pkt_alu_reg_block::type_id::create("reg_model");
44 reg_model.build();
45 reg_adapter = apb_reg_adapter::type_id::create("reg_adapter");
46 reg_predictor = uvm_reg_predictor#(apb_item)::type_id::create("reg_predictor", this);
47 
48 if (cfg.enable_scoreboard) scb = alu_scoreboard::type_id::create("scb", this);
49 if (cfg.enable_coverage) cov = alu_cov_subscriber::type_id::create("cov", this);
50 
51 apb_domain = new("apb_domain");
52 hb_obj = new("hb_obj");
53 hb = new("hb", this, hb_obj);
54 hb_trig = new("hb_trig");
55 endfunction
56 
57 virtual function void connect_phase(uvm_phase phase);
58 super.connect_phase(phase);
59 if (data_agent.seqr != null) v_seqr.alu_sqr = data_agent.seqr;
60 if (apb_agt.seqr != null) v_seqr.apb_sqr = apb_agt.seqr;
61 
62 if (apb_agt.seqr != null) begin
63 reg_model.apb_map.set_sequencer(apb_agt.seqr, reg_adapter);
64 end
65 reg_model.apb_map.set_auto_predict(0);
66 reg_predictor.map = reg_model.apb_map;
67 reg_predictor.adapter = reg_adapter;
68 apb_agt.ap.connect(reg_predictor.bus_in);
69 
70 if (cfg.enable_scoreboard) begin
71 scb.reg_model = reg_model;
72 data_agent.req_ap.connect(scb.req_export);
73 data_agent.rsp_ap.connect(scb.rsp_export);
74 // Connect TLM-1 monitor -> TLM-2 bridge initiator socket -> ref_model target socket
75 data_agent.req_ap.connect(tlm2_bridge.req_imp);
76 tlm2_bridge.nb_init_socket.connect(scb.ref_model.nb_target_socket);
77 end
78 if (cfg.enable_coverage) begin
79 data_agent.req_ap.connect(cov.analysis_export);
80 end
81 endfunction
82endclass
