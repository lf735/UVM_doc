25/*
26About: uvm_examples/uvm_tlm/bidir
27This example will illustrate how to create two threads one is the master and the other is the slave. The two threads are completely independent, each one of the two threads has two types of ports interfaces puts and gets.
28 
29The communication between the two channel will be done through a uvm_tlm_req_rsp channel , which can handle the dual Request and Response of each thread
30 
31This example will use The uvm_tlm_req_rsp channel, transaction-level ports, and the messaging facilities as part of the UVM library.
32*/

40//----------------------------------------------------------------------
41// class master
42//----------------------------------------------------------------------
43class master extends uvm_component;
44 
45 uvm_blocking_put_port #(int) req_port;
46 uvm_blocking_get_port #(int) rsp_port;
47 
48 function new(string name, uvm_component parent = null);
49 super.new(name, parent);
50 req_port = new("req_port", this);
51 rsp_port = new("rsp_port", this);
52 endfunction
53
54 task run_phase(uvm_phase phase);
55 fork
56 request_process;
57 response_process;
58 join
59 endtask
60 
61 task request_process;
62 
63 string request_str;
64
65 for(int i = 0; i < 10; i++) begin
66 $sformat(request_str, "%d", i);
67 uvm_report_info("sending request ", request_str);
68 req_port.put(i);
69 end
70
71 endtask // request_process
72
73 task response_process;
74 
75 int response;
76 string response_str;
77
78 forever begin
79 rsp_port.get(response);
80 $sformat(response_str, "%d", response);
81 uvm_report_info("recieving response", response_str);
82 end
83 endtask
84 
85endclass // master
86 
87 
88//----------------------------------------------------------------------
89// class slave
90//----------------------------------------------------------------------
91class slave extends uvm_component;
92 
93 uvm_blocking_get_port #(int) req_port;
94 uvm_blocking_put_port #(int) rsp_port;
95
96 function new(string name, uvm_component parent = null);
97 super.new(name, parent);
98 req_port = new("req_port", this);
99 rsp_port = new("rsp_port", this);
100 endfunction // new
101
102 task run_phase(uvm_phase phase);
103 
104 int request, response;
105 string request_str, response_str;
106
107 forever begin
108 req_port.get(request);
109 $sformat(request_str, "%d", request);
110 uvm_report_info("recieving request ", request_str);
111 
112 response = request;
113
114 $sformat(response_str, "%d", response);
115 uvm_report_info("sending response ", response_str);
116 
117 rsp_port.put(response);
118
119 end // forever begin
120 endtask
121 
122endclass

124//----------------------------------------------------------------------
125// class bidir_env
126//----------------------------------------------------------------------
127class bidir_env extends uvm_env;
128 
129 master m;
130 slave s;
131 uvm_tlm_req_rsp_channel #(int) req_rsp;
132 
133 function new(string name, uvm_component parent);
134 super.new(name, parent);
135 m = new("master", this);
136 s = new("slave", this);
137 req_rsp = new("req_rsp_channel", this);
138 endfunction
139 
140 function void connect_phase(uvm_phase phase);
141 m.req_port.connect(req_rsp.blocking_put_request_export);
142 m.rsp_port.connect(req_rsp.blocking_get_response_export);
143 s.req_port.connect(req_rsp.blocking_get_request_export);
144 s.rsp_port.connect(req_rsp.blocking_put_response_export);
145 endfunction
146 
147 task run_phase(uvm_phase phase);
148 phase.raise_objection(this);
149 #10;
150 phase.drop_objection(this);
151 endtask
152
153endclass
