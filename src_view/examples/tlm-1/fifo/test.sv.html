24/*
25About: uvm_tlm_fifo
26 
27This test case supposed to test a producer consumer connection through a uvm_tlm_fifo of 10 levels.
28 
29 
30 
31Walk through the test:
32Inside a module, there will be two threads established, the first one the *producer* will put a packet of an integer to put port, the second one *consumer* will get this packet through a get port. The put and get ports will be connected to the uvm_tlm_fifo exports.
33 
34the method *fifo.used* will be used here to display which fifo level is used.
35 
36*/

40module test;
41 import uvm_pkg::*;
42 `include "uvm_macros.svh"
43 
44 class packet;
45 int i;
46 function new(int v); i=v; endfunction
47 endclass
48 
49 class producer extends uvm_component;
50 uvm_put_port #(packet) data_out;
51 function new(string name, uvm_component parent);
52 super.new(name,parent);
53 data_out = new("data_out", this);
54 endfunction
55 task run_phase(uvm_phase phase);
56 packet p, pp;
57 #1 p=new(0);
58 while(data_out.try_put(p)) begin
59 $display("%0t: put data %0d", $time, p.i);
60 #10 p = new(p.i+1);
61 end
62 $display("try_put status return: %0d", p.i);
63 $display("%0t: do a blocking put", $time);
64 data_out.put(p);
65 $display("%0t: blocking put succeeded", $time);
66 endtask
67 endclass
68 class consumer extends uvm_component;
69 uvm_get_port #(packet) data_in;
70 function new(string name, uvm_component parent);
71 super.new(name,parent);
72 data_in = new("data_in", this);
73 endfunction
74 task run_phase(uvm_phase phase);
75 packet p;
76 #100; // fifo will fill up
77 $display("%0t: getting one", $time);
78 data_in.get(p);
79 $display("%0t: received data %0d", $time, p.i);
80 #100; // let the blocking put succeed
81 while(data_in.try_get(p)) begin
82 $display("%0t: received data %0d", $time, p.i);
83 #10;
84 end
85 endtask
86 endclass

89 class test extends uvm_test;
90 
91 `uvm_component_utils(test)
92 
93 function new(string name = "", uvm_component parent = null);
94 super.new(name, parent);
95 endfunction
96 
97 task run_phase(uvm_phase phase);
98 phase.raise_objection(null);
99 #5us;
100 phase.drop_objection(null);
101 endtask
102 endclass
103 
104 producer prod = new("prod", null);
105 consumer cons = new("cons", null);
106 uvm_tlm_fifo #(packet) fifo = new("fifo", null, 10);
107 
108 initial begin
109 prod.data_out.connect(fifo.put_export);
110 cons.data_in.connect(fifo.get_export);
111 
112 fork
113 run_test("test");
114 repeat(30) begin
115 $display("%0t: FIFO level %0d of %0d", $time, fifo.used(), fifo.size());
116 #10;
117 end
118 join
119 end
