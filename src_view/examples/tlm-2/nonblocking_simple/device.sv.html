24 
25 `uvm_component_utils(device)

27 function new(string name = "device", uvm_component parent = null);
28 super.new(name, parent);
29 sock = new("sock", this);
30 endfunction

32 byte data[];
33 
34 // Forward path
35 function uvm_tlm_sync_e nb_transport_fw(usb_xfer xfer,
36 ref usb_tlm_phase ph,
37 input uvm_tlm_time delay);
38 
39 `uvm_info("USB/DEV/FWD", $sformatf("%s @%0d: %s",
40 ph.name(), delay.get_realtime(1ns),
41 xfer.convert2string()), UVM_LOW)
42 
43 // This device?
44 if (xfer.addr != 7'h7A) return UVM_TLM_COMPLETED;
45 
46 // Valid endpoint in the device?
47 if (xfer.endp != 4'h5) return UVM_TLM_COMPLETED;
48
49 case (xfer.kind)
50 
51 usb_xfer::OUT: begin
52 case (ph)
53 USB_TLM_TOKEN: begin
54 return UVM_TLM_ACCEPTED;
55 end
56 
57 USB_TLM_DATA: begin
58 data = xfer.data;
59 
60 // Could complete transfer early here
61 fork: out_ack
62 automatic usb_xfer xf = xfer;
63 automatic uvm_tlm_time dl = delay;
64 begin
65 usb_tlm_phase ph = USB_TLM_HANDSHAKE;
66 #100;
67 xf.status = usb_xfer::ACK;
68 assert(sock.nb_transport_bw(xf, ph, dl) ==
69 UVM_TLM_COMPLETED);
70 
71 `uvm_info("USB/DEV/OUT/DONE", xf.convert2string(),
72 UVM_NONE);
73
74 end
75 join_none
76 
77 return UVM_TLM_ACCEPTED;
78 end
79
80 endcase
81 end
82 
83 usb_xfer::IN: begin
84 case (ph)
85 USB_TLM_TOKEN: begin
86 // Could return the data early here
87 fork: in_data
88 automatic usb_xfer xf = xfer;
89 automatic uvm_tlm_time dl = delay;
90 begin
91 usb_tlm_phase ph = USB_TLM_DATA;
92 #150;
93 begin
94 byte tdata[2]='{'hAB, 'hCD};
95 xf.data = tdata;
96 end
97 void'(sock.nb_transport_bw(xf, ph, dl));
98 end
99 join_none
100 
101 return UVM_TLM_ACCEPTED;
102 end
103 
104 USB_TLM_HANDSHAKE: begin
105 `uvm_info("USB/DEV/IN/DONE", xfer.convert2string(),
106 UVM_NONE);
107
108 return UVM_TLM_COMPLETED;
109 end
110
111 endcase
112 end
113 endcase
114
115 return UVM_TLM_COMPLETED;
116 endfunction
