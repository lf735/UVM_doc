36class apb_monitor extends uvm_monitor;
37 virtual apb_if.passive sigs;
38 
39 uvm_analysis_port#(apb_rw) ap;
40 apb_config cfg;
41 
42 `uvm_component_utils(apb_monitor)
43 
44 function new(string name, uvm_component parent = null);
45 super.new(name, parent);
46 ap = new("ap", this);
47 endfunction: new
48 
49 virtual function void build_phase(uvm_phase phase);
50 apb_agent agent;
51 if ($cast(agent, get_parent()) && agent != null) begin
52 sigs = agent.vif;
53 end
54 else begin
55 virtual apb_if tmp;
56 if (!uvm_config_db#(apb_vif)::get(this, "", "vif", tmp)) begin
57 `uvm_fatal("APB/MON/NOVIF", "No virtual interface specified for this monitor instance")
58 end
59 sigs = tmp;
60 end
61 endfunction
62 
63 virtual task run_phase(uvm_phase phase);
64 super.run_phase(phase);
65 forever begin
66 apb_rw tr;
67
68 // Wait for a SETUP cycle
69 do begin
70 @ (this.sigs.pck);
71 end
72 while (this.sigs.pck.psel !== 1'b1 ||
73 this.sigs.pck.penable !== 1'b0);
74 
75 tr = apb_rw::type_id::create("tr", this);
76
77 tr.kind = (this.sigs.pck.pwrite) ? apb_rw::WRITE : apb_rw::READ;
78 tr.addr = this.sigs.pck.paddr;
79 
80 @ (this.sigs.pck);
81 if (this.sigs.pck.penable !== 1'b1) begin
82 `uvm_error("APB", "APB protocol violation: SETUP cycle not followed by ENABLE cycle");
83 end
84 tr.data = (tr.kind == apb_rw::READ) ? this.sigs.pck.prdata :
85 this.sigs.pck.pwdata;
86 
87 trans_observed(tr);
88 `uvm_do_callbacks(apb_monitor,apb_monitor_cbs,trans_observed(this,tr))
89 
90 ap.write(tr);
91 end
92 endtask: run_phase
93 
94 virtual protected task trans_observed(apb_rw tr);
95 endtask
