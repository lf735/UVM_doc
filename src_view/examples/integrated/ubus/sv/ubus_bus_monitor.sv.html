97 // The virtual interface used to view HDL signals.

102 
103 // The following two bits are used to control whether checks and coverage are
104 // done both in the bus monitor class and the interface.
105 bit checks_enable = 1;
106 bit coverage_enable = 1;
107 
108 // Analysis ports for the item_collected and state notifier.
109 uvm_analysis_port #(ubus_transfer) item_collected_port;
110 uvm_analysis_port #(ubus_status) state_port;
111 
112 // The state of the ubus
113 protected ubus_status status;
114 
115 // The following property is used to store slave address map
116 protected slave_address_map_info slave_addr_map[string];
117 
118 // The following property holds the transaction information currently
119 // being captured (by the collect_address_phase and data_phase methods).
120 protected ubus_transfer trans_collected;
121 
122 // Events needed to trigger covergroups
123 protected event cov_transaction;
124 protected event cov_transaction_beat;

126 // Fields to hold trans data and wait_state. No coverage of dynamic arrays.
127 protected bit [15:0] addr;
128 protected bit [7:0] data;

131 // Transfer collected covergroup
132 covergroup cov_trans @cov_transaction;
133 option.per_instance = 1;
134 trans_start_addr : coverpoint trans_collected.addr {
135 option.auto_bin_max = 16; }
136 trans_dir : coverpoint trans_collected.read_write;
137 trans_size : coverpoint trans_collected.size {
138 bins sizes[] = {1, 2, 4, 8};
139 illegal_bins invalid_sizes = default; }
140 trans_addrXdir : cross trans_start_addr, trans_dir;
141 trans_dirXsize : cross trans_dir, trans_size;
142 endgroup : cov_trans

144 // Transfer collected data covergroup

150 beat_data : coverpoint data {
151 option.auto_bin_max = 8; }
152 beat_wait : coverpoint wait_state {
153 bins waits[] = { [0:9] };
154 bins others = { [10:$] }; }
155 beat_addrXdir : cross beat_addr, beat_dir;
156 beat_addrXdata : cross beat_addr, beat_data;

187 function void build_phase(uvm_phase phase);
188 if(!uvm_config_db#(virtual ubus_if)::get(this, "", "vif", vif))
189 `uvm_fatal("NOVIF",{"virtual interface must be set for: ",get_full_name(),".vif"});
190 endfunction: build_phase
191 
192 // run phase
193 task run_phase(uvm_phase phase);
194 fork
195 observe_reset();
196 collect_transactions();
197 join
198 endtask : run_phase
199 
200 // observe_reset
201 task observe_reset();
202 fork
203 forever begin
204 @(posedge vif.sig_reset);
205 status.bus_state = RST_START;
206 state_port.write(status);
207 end
208 forever begin
209 @(negedge vif.sig_reset);
210 status.bus_state = RST_STOP;
211 state_port.write(status);
212 end
213 join
214 endtask : observe_reset
215 
216 // collect_transactions
217 virtual protected task collect_transactions();
218 forever begin
219 collect_arbitration_phase();
220 collect_address_phase();
221 collect_data_phase();
222 `uvm_info(get_type_name(),$sformatf("Transfer collected :\n%s",
223 trans_collected.sprint()), UVM_HIGH)
224 if (checks_enable)
225 perform_transfer_checks();
226 if (coverage_enable)
227 perform_transfer_coverage();
228 item_collected_port.write(trans_collected);
229 end
230 endtask : collect_transactions
231 
232 // collect_arbitration_phase
233 task collect_arbitration_phase();
234 string tmpStr;
235 @(posedge vif.sig_clock iff (vif.sig_grant != 0));
236 status.bus_state = ARBI;
237 state_port.write(status);
238 void'(this.begin_tr(trans_collected));
239 // Check which grant is asserted to determine which master is performing
240 // the transfer on the bus.
241 for (int j = 0; j <= 15; j++) begin
242 if (vif.sig_grant[j] === 1) begin
243 $sformat(tmpStr,"masters[%0d]", j);
244 trans_collected.master = tmpStr;
245 break;
246 end
247 end
248 endtask : collect_arbitration_phase
249 
250 // collect_address_phase
251 task collect_address_phase();
252 @(posedge vif.sig_clock);
253 trans_collected.addr = vif.sig_addr;
254 case (vif.sig_size)
255 2'b00 : trans_collected.size = 1;
256 2'b01 : trans_collected.size = 2;
257 2'b10 : trans_collected.size = 4;
258 2'b11 : trans_collected.size = 8;
259 endcase
260 trans_collected.data = new[trans_collected.size];
261 case ({vif.sig_read,vif.sig_write})
262 2'b00 : begin
263 trans_collected.read_write = NOP;
264 status.bus_state = NO_OP;
265 state_port.write(status);
266 end
267 2'b10 : begin
268 trans_collected.read_write = READ;
269 status.bus_state = ADDR_PH;
270 state_port.write(status);
271 end
272 2'b01 : begin
273 trans_collected.read_write = WRITE;
274 status.bus_state = ADDR_PH;
275 state_port.write(status);
276 end
277 2'b11 : begin
278 status.bus_state = ADDR_PH_ERROR;
279 state_port.write(status);
280 if (checks_enable)
281 `uvm_error(get_type_name(),
282 "Read and Write true at the same time")
283 end
284 endcase
285 endtask : collect_address_phase
286 
287 // collect_data_phase
288 task collect_data_phase();
289 int i;
290 if (trans_collected.read_write != NOP) begin
291 check_which_slave();
292 for (i = 0; i < trans_collected.size; i++) begin
293 status.bus_state = DATA_PH;
294 state_port.write(status);
295 @(posedge vif.sig_clock iff vif.sig_wait === 0);
296 trans_collected.data[i] = vif.sig_data;
297 end
298 num_transactions++;
299 this.end_tr(trans_collected);
300 end
301 endtask : collect_data_phase
302 
303 // check_which_slave
304 function void check_which_slave();
305 string slave_name;
306 bit slave_found;
307 slave_found = 1'b0;
308 if(slave_addr_map.first(slave_name))
309 do begin
310 if (slave_addr_map[slave_name].get_min_addr() <= trans_collected.addr
311 && trans_collected.addr <= slave_addr_map[slave_name].get_max_addr())
312 begin
313 trans_collected.slave = slave_name;
314 slave_found = 1'b1;
315 end
316 if (slave_found == 1'b1)
317 break;
318 end
319 while (slave_addr_map.next(slave_name));
320 assert(slave_found) else begin
321 `uvm_error(get_type_name(),
322 $sformatf("Master attempted a transfer at illegal address 16'h%0h",
323 trans_collected.addr))
324 end
325 endfunction : check_which_slave
326 
327 // perform_transfer_checks
328 function void perform_transfer_checks();
329 check_transfer_size();
330 check_transfer_data_size();
331 endfunction : perform_transfer_checks
332 
333 // check_transfer_size
334 function void check_transfer_size();
335 if (trans_collected.read_write != NOP) begin
336 assert_transfer_size : assert(trans_collected.size == 1 ||
337 trans_collected.size == 2 || trans_collected.size == 4 ||
338 trans_collected.size == 8) else begin
339 `uvm_error(get_type_name(),
340 "Invalid transfer size!")
341 end
342 end
343 endfunction : check_transfer_size
344 
345 // check_transfer_data_size
346 function void check_transfer_data_size();
347 if (trans_collected.size != trans_collected.data.size())
348 `uvm_error(get_type_name(),
349 "Transfer size field / data size mismatch.")
350 endfunction : check_transfer_data_size
351 
352 // perform_transfer_coverage
353 function void perform_transfer_coverage();
354 if (trans_collected.read_write != NOP) begin
355 -> cov_transaction;
356 for (int unsigned i = 0; i < trans_collected.size; i++) begin
357 addr = trans_collected.addr + i;
358 data = trans_collected.data[i];
359 //wait_state = trans_collected.wait_state[i];
360 -> cov_transaction_beat;
361 end
362 end
363 endfunction : perform_transfer_coverage
