28class ubus_master_monitor extends uvm_monitor;
29 
30 // This property is the virtual interfaced needed for this component to drive
31 // and view HDL signals.
32 protected virtual ubus_if vif;
33 
34 // Master Id
35 protected int master_id;
36 
37 // The following two bits are used to control whether checks and coverage are
38 // done both in the monitor class and the interface.
39 bit checks_enable = 1;
40 bit coverage_enable = 1;
41 
42 uvm_analysis_port #(ubus_transfer) item_collected_port;
43 
44 // The following property holds the transaction information currently
45 // begin captured (by the collect_address_phase and data_phase methods).
46 protected ubus_transfer trans_collected;
47 
48 // Fields to hold trans addr, data and wait_state.
49 protected bit [15:0] addr;
50 protected bit [7:0] data;
51 protected int unsigned wait_state;

53 // Transfer collected covergroup
54 covergroup cov_trans;
55 option.per_instance = 1;
56 trans_start_addr : coverpoint trans_collected.addr {
57 option.auto_bin_max = 16; }
58 trans_dir : coverpoint trans_collected.read_write;
59 trans_size : coverpoint trans_collected.size {
60 bins sizes[] = {1, 2, 4, 8};
61 illegal_bins invalid_sizes = default; }
62 trans_addrXdir : cross trans_start_addr, trans_dir;
63 trans_dirXsize : cross trans_dir, trans_size;
64 endgroup : cov_trans

66 // Transfer collected beat covergroup
67 covergroup cov_trans_beat;
68 option.per_instance = 1;
69 beat_addr : coverpoint addr {
70 option.auto_bin_max = 16; }
71 beat_dir : coverpoint trans_collected.read_write;
72 beat_data : coverpoint data {
73 option.auto_bin_max = 8; }
74 beat_wait : coverpoint wait_state {

88 // new - constructor
89 function new (string name, uvm_component parent);
90 super.new(name, parent);
91 cov_trans = new();
92 cov_trans.set_inst_name({get_full_name(), ".cov_trans"});
93 cov_trans_beat = new();
94 cov_trans_beat.set_inst_name({get_full_name(), ".cov_trans_beat"});
95 trans_collected = new();
96 item_collected_port = new("item_collected_port", this);
97 endfunction : new

99 function void build_phase(uvm_phase phase);
100 super.build_phase(phase);
101 if(!uvm_config_db#(virtual ubus_if)::get(this, "", "vif", vif))
102 `uvm_fatal("NOVIF",{"virtual interface must be set for: ",get_full_name(),".vif"});
103 endfunction: build_phase
104 
105 // run phase
106 virtual task run_phase(uvm_phase phase);
107 `uvm_info({get_full_name()," MASTER ID"},$sformatf(" = %0d",master_id),UVM_MEDIUM)
108 fork
109 collect_transactions();
110 join
111 endtask : run_phase
112 
113 // collect_transactions
114 virtual protected task collect_transactions();
115 forever begin
116 @(posedge vif.sig_clock);
117 if (m_parent != null)
118 trans_collected.master = m_parent.get_name();
119 collect_arbitration_phase();
120 collect_address_phase();
121 collect_data_phase();
122 `uvm_info(get_full_name(), $sformatf("Transfer collected :\n%s",
123 trans_collected.sprint()), UVM_MEDIUM)
124 if (checks_enable)
125 perform_transfer_checks();
126 if (coverage_enable)
127 perform_transfer_coverage();
128 item_collected_port.write(trans_collected);
129 end
130 endtask : collect_transactions
131 
132 // collect_arbitration_phase
133 virtual protected task collect_arbitration_phase();
134 @(posedge vif.sig_request[master_id]);
135 @(posedge vif.sig_clock iff vif.sig_grant[master_id] === 1);
136 void'(this.begin_tr(trans_collected));
137 endtask : collect_arbitration_phase
138 
139 // collect_address_phase
140 virtual protected task collect_address_phase();
141 @(posedge vif.sig_clock);
142 trans_collected.addr = vif.sig_addr;
143 case (vif.sig_size)
144 2'b00 : trans_collected.size = 1;
145 2'b01 : trans_collected.size = 2;
146 2'b10 : trans_collected.size = 4;
147 2'b11 : trans_collected.size = 8;
148 endcase
149 trans_collected.data = new[trans_collected.size];
150 case ({vif.sig_read,vif.sig_write})
151 2'b00 : trans_collected.read_write = NOP;
152 2'b10 : trans_collected.read_write = READ;
153 2'b01 : trans_collected.read_write = WRITE;
154 endcase
155 endtask : collect_address_phase
156 
157 // collect_data_phase
158 virtual protected task collect_data_phase();
159 int i;
160 if (trans_collected.read_write != NOP)
161 for (i = 0; i < trans_collected.size; i++) begin
162 @(posedge vif.sig_clock iff vif.sig_wait === 0);
163 trans_collected.data[i] = vif.sig_data;
164 end
165 this.end_tr(trans_collected);
166 endtask : collect_data_phase
167 
168 // perform_transfer_checks
169 virtual protected function void perform_transfer_checks();
170 check_transfer_size();
171 check_transfer_data_size();
172 endfunction : perform_transfer_checks
173 
174 // check_transfer_size
175 virtual protected function void check_transfer_size();
176 assert_transfer_size : assert(trans_collected.size == 1 ||
177 trans_collected.size == 2 || trans_collected.size == 4 ||
178 trans_collected.size == 8) else begin
179 `uvm_error(get_type_name(),
180 "Invalid transfer size!")
181 end
182 endfunction : check_transfer_size
183 
184 // check_transfer_data_size
185 virtual protected function void check_transfer_data_size();
186 if (trans_collected.size != trans_collected.data.size())
187 `uvm_error(get_type_name(),
188 "Transfer size field / data size mismatch.")
189 endfunction : check_transfer_data_size
190 
191 // perform_transfer_coverage
192 virtual protected function void perform_transfer_coverage();
193 cov_trans.sample();
194 for (int unsigned i = 0; i < trans_collected.size; i++) begin
195 addr = trans_collected.addr + i;
196 data = trans_collected.data[i];
197//Wait state is not currently monitored
198// wait_state = trans_collected.wait_state[i];
199 cov_trans_beat.sample();
200 end
201 endfunction : perform_transfer_coverage
202 
203 virtual function void report_phase(uvm_phase phase);
204 `uvm_info(get_full_name(),$sformatf("Covergroup 'cov_trans' coverage: %2f",
205 cov_trans.get_inst_coverage()),UVM_LOW)
206 endfunction
