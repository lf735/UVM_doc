28class ubus_slave_driver extends uvm_driver #(ubus_transfer);
29 
30 // The virtual interface used to drive and view HDL signals.
31 protected virtual ubus_if vif;
32 
33 // Provide implementations of virtual methods such as get_type_name and create
34 `uvm_component_utils(ubus_slave_driver)
35 
36 // new - constructor
37 function new (string name, uvm_component parent);
38 super.new(name, parent);
39 endfunction : new
40 
41 function void build_phase(uvm_phase phase);
42 if(!uvm_config_db#(virtual ubus_if)::get(this, "", "vif", vif))
43 `uvm_fatal("NOVIF",{"virtual interface must be set for: ",get_full_name(),".vif"});
44 endfunction: build_phase
45 
46 // run phase
47 virtual task run_phase(uvm_phase phase);
48 fork
49 get_and_drive();
50 reset_signals();
51 join
52 endtask : run_phase
53 
54 // get_and_drive
55 virtual protected task get_and_drive();
56 @(negedge vif.sig_reset);
57 forever begin
58 @(posedge vif.sig_clock);
59 seq_item_port.get_next_item(req);
60 respond_to_transfer(req);
61 seq_item_port.item_done();
62 end
63 endtask : get_and_drive
64 
65 // reset_signals
66 virtual protected task reset_signals();
67 forever begin
68 @(posedge vif.sig_reset);
69 vif.sig_error <= 1'bz;
70 vif.sig_wait <= 1'bz;
71 vif.rw <= 1'b0;
72 end
73 endtask : reset_signals
74 
75 // respond_to_transfer
76 virtual protected task respond_to_transfer(ubus_transfer resp);
77 if (resp.read_write != NOP)
78 begin
79 vif.sig_error <= 1'b0;
80 for (int i = 0; i < resp.size; i++)
81 begin
82 case (resp.read_write)
83 READ : begin
84 vif.rw <= 1'b1;
85 vif.sig_data_out <= resp.data[i];
86 end
87 WRITE : begin
88 end
89 endcase
90 if (resp.wait_state[i] > 0) begin
91 vif.sig_wait <= 1'b1;
92 repeat (resp.wait_state[i])
93 @(posedge vif.sig_clock);
94 end
95 vif.sig_wait <= 1'b0;
96 @(posedge vif.sig_clock);
97 resp.data[i] = vif.sig_data;
98 end
99 vif.rw <= 1'b0;
100 vif.sig_wait <= 1'bz;
101 vif.sig_error <= 1'bz;
102 end
103 endtask : respond_to_transfer
104 
105endclass : ubus_slave_driver
