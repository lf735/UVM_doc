44 function void build_phase(uvm_phase phase);
45 void'(uvm_config_db#(int)::get(this, "", "n_obs_thresh", n_obs_thresh));

48 function void write_sym_sb_expected(vip_tr tr);
49 `uvm_info("SB/EXP", $sformatf("Expected: 0x%h", tr.chr), UVM_MEDIUM)
50 m_sb.push_back(tr.chr);
51 endfunction
52 
53 function void write_sym_sb_observed(vip_tr tr);
54 bit [7:0] exp;
55
56 `uvm_info("SB/OBS", $sformatf("Observed: 0x%h", tr.chr), UVM_MEDIUM)
57 
58 exp = m_sb.pop_front();
59 if (tr.chr !== exp) begin
60 `uvm_error("SB/MISMTCH",
61 $sformatf("Symbol 0x%h observed instead of expected 0x%h",
62 tr.chr, exp));
63 end
64 m_n_obs++;
65 endfunction

69 m_n_obs = 0;
70 m_sb.delete();

74 phase.raise_objection(this, "Have not checked enough data");
