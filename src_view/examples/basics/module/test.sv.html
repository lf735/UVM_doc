61class myunit extends uvm_component;
62 lower l1;
63 lower l2;
64 int a[];
65 
66 function new (string name, uvm_component parent);
67 super.new(name, parent);
68 uvm_config_string::set(this, "l1", "str", "hi");
69 uvm_config_int::set(this, "*", "data", 'h100);
70 l1 = new ("l1", this);
71 l2 = new ("l2", this);
72 l1.data = 'h30;
73 l2.data = 'h40;
74 a = new[5]; for(int i=0; i<5;++i) a[i] = i*i;
75 endfunction
76 
77 task run_phase(uvm_phase phase);
78 //Check config from initial block
79 if(l1.data != 55)
80 `uvm_error("BADCFG", $sformatf("Expected l1.data = 55, got %0d", l1.data))
81 if(l2.data != 101)
82 `uvm_error("BADCFG", $sformatf("Expected l2.data = 101, got %0d", l2.data))
83 if(l1.str != l2.str && l1.str != "hi")
84 `uvm_error("BADCFG", $sformatf("Expected l1.str = \"hi\" and l2.str = \"hi\", got l1.str = \"%s\" and l2.str = \"%s\"", l1.str, l2.str))
85 phase.raise_objection(this);
86 #10 $display("%0t: %s HI", $time, get_full_name());
87 phase.drop_objection(this);
88 endtask
89 
90 function string get_type_name();
91 return "myunit";
92 endfunction

94 function void do_print(uvm_printer printer);
95 printer.print_array_header("a", a.size());
96 for(int i=0; i<a.size(); ++i)
97 printer.print_field( $sformatf("a[%0d]", i), a[i], 32, UVM_HEX, "[");
98 printer.print_array_footer();
