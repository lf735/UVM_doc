34// the same names so we can reuse the generic any_driver for

65class reg2apb_adapter extends uvm_reg_adapter;
66 
67 `uvm_object_utils(reg2apb_adapter)
68 
69 virtual function uvm_sequence_item reg2bus(const ref uvm_reg_bus_op rw);
70 apb_item apb = apb_item::type_id::create("apb_item");
71 apb.read = (rw.kind == UVM_READ) ? 1 : 0;
72 apb.addr = rw.addr;
73 apb.data = rw.data;
74 return apb;
75 endfunction
76 
77 virtual function void bus2reg(uvm_sequence_item bus_item, ref uvm_reg_bus_op rw);
78 apb_item apb;
79 if (!$cast(apb,bus_item)) begin
80 `uvm_fatal("NOT_APB_TYPE","Provided bus_item is not of the correct type")
81 return;
82 end
83 rw.status = UVM_IS_OK;
84 rw.kind = apb.read ? UVM_READ : UVM_WRITE;
85 rw.addr = apb.addr;
86 rw.data = apb.data;
87 endfunction

90 function new(string name="reg2apb_adapter");
91 super.new(name);
92 endfunction

94endclass

128class my_apb_driver extends any_driver #(apb_item);
129 `uvm_component_utils(my_apb_driver)
130 function new(string name = "apb_driver", uvm_component parent=null);
131 super.new(name, parent);
132 endfunction: new
133 virtual task pre_req(apb_item req);
134 // remap so for APB: regA is at addr 1, regX is at addr 0
135 // for WSH: shared regX is also at addr 0, regW is at addr 'h10
136 if (req.addr == 'h10) // regX bus addr => mem[0]
137 req.addr = 'h0;
138 else if (req.addr == 'h0) // regA bus addr => mem[1]
139 req.addr = 'h1;
140 endtask

173 virtual function void connect_phase(uvm_phase phase);
174 reg2apb_adapter reg2apb = new;
175 reg2wsh_adapter reg2wsh = new;
176 model.APB.set_sequencer(apb.sqr, reg2apb);
177 model.APB.set_auto_predict();
178 model.WSH.set_sequencer(wsh.sqr, reg2wsh);
179 model.WSH.set_auto_predict();
180 apb.drv.base_addr = 'h20;
