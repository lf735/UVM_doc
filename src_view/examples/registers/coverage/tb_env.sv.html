26class dut;
27 static bit [3:0] Ra_F1 = 0;
28 static bit [3:0] Ra_F2 = 0;
29 static bit [3:0] Rb_F1 = 0;
30 static bit [3:0] Rb_F2 = 0;
31 static bit [7:0] M[1024];
32 
33 static task rw(reg_rw rw);
34 casez (rw.addr)
35 'h0000: // Ra
36 if (rw.read) rw.data = {16'h00, Ra_F2, Ra_F1};
37 else begin
38 if (rw.byte_en[0]) Ra_F1 = rw.data[ 7:0];
39 if (rw.byte_en[0]) Ra_F2 = rw.data[15:8];
40 end
41 'h0100: // Rb
42 if (rw.read) rw.data = {16'h00, Rb_F2, Rb_F1};
43 else begin
44 if (rw.byte_en[0]) Rb_F1 = rw.data[ 7:0];
45 if (rw.byte_en[0]) Rb_F2 = rw.data[15:8];
46 end
47 'b001000??_????????: // M
48 if (rw.read) rw.data = {24'h00, M[rw.addr[9:0]]};
49 else begin
50 if (rw.byte_en[0]) M[rw.addr[9:0]] = rw.data[ 7:0];
51 end
52 endcase
53 #1;
54 // $write("%s\n", rw.convert2string());
55 endtask
56endclass

71 virtual function void build_phase(uvm_phase phase);
72 regmodel = block_B::type_id::create("regmodel");
73 regmodel.build();
74 regmodel.lock_model();
75 
76 bus = reg_agent#(dut)::type_id::create("bus", this);
77 predict = uvm_reg_predictor#(reg_rw)::type_id::create("predict", this);
78 endfunction: build_phase
79 
80 virtual function void connect_phase(uvm_phase phase);
81 reg2rw_adapter reg2rw = new("reg2rw");
82 regmodel.default_map.set_sequencer(bus.sqr, reg2rw);
83 
84 predict.map = regmodel.default_map;
85 predict.adapter = reg2rw;
86 bus.mon.ap.connect(predict.bus_in);
87 regmodel.default_map.set_auto_predict(0);
88 endfunction
