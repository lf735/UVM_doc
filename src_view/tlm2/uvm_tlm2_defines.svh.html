21//----------------------------------------------------------------------
22// Title: Interface Masks
23//
24// Each of the following macros is a mask that identifies which
25// interfaces a particular port requires or export provides. The
26// interfaces are identified by bit position and can be OR'ed together
27// for combination ports/exports. The mask is used to do run-time
28// interface type checking of port/export connections.
29//----------------------------------------------------------------------
30 
31// MACRO: `UVM_TLM_NB_FW_MASK
32//
33// Define Non blocking Forward mask onehot assignment = 'b001
34`define UVM_TLM_NB_FW_MASK (1<<0)
35 
36// MACRO: `UVM_TLM_NB_BW_MASK
37//
38// Define Non blocking backward mask onehot assignment = 'b010
39`define UVM_TLM_NB_BW_MASK (1<<1)
40 
41// MACRO: `UVM_TLM_B_MASK
42//
43// Define blocking mask onehot assignment = 'b100
44`define UVM_TLM_B_MASK (1<<2)
