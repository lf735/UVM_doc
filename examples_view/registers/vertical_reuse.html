Vertical Register Model Reuse

Demonstrates reusing block-level register sequences and register models unchanged inside system-level testbenches with hierarchical address offsets.

Key UVM APIs: uvm_reg_block default_map add_submap

Testbench Architecture Topology

system sys_reg_model

23module blk_dut #(int BASE_ADDR='h0) (apb_if apb,
24 input bit rst);
25 
26reg [7:0] R;
27 
28reg [31:0] pr_data;
29 
30wire in_range;
31 
32wire [31:0] pr_addr;
33 
34assign in_range = (apb.paddr - BASE_ADDR) < 'h100;
35assign pr_addr = apb.paddr - BASE_ADDR;
36 
37assign apb.prdata = (apb.psel && apb.penable && !apb.pwrite && in_range) ? pr_data : 'z;
38 
39 
40always @ (posedge apb.pclk)
41 begin
42 if (rst) begin
43 R <= 'h00;
44 pr_data <= 32'h0;
45 end
46 else begin
47 
48 // Wait for a SETUP+READ or ENABLE+WRITE cycle
49 if (apb.psel == 1'b1 && apb.penable == apb.pwrite) begin
50 pr_data <= 32'h0;
51 if (apb.pwrite) begin
52 casex (pr_addr)
53 32'h00000000:
54 R <= apb.pwdata[7:0];
55 32'h00000001:
56 casez (apb.pwdata[1:0])
57 2'b01: R++;
58 2'b10: R--;
59 2'b11: R <= 0;
60 endcase
61 endcase
62 end
63 else begin
64 casex (pr_addr)
65 32'h00000000: pr_data <= {24'h0, R};
66 default: pr_data <= 32'h0;
67 endcase
68 #1;
69 end
70 #0;
71 end
72 end
73end
74 
75endmodule

24class blk_R_test_seq extends uvm_reg_sequence;
25 
26 `uvm_object_utils(blk_R_test_seq)
27 
28 function new(string name = "blk_R_test_seq");
29 super.new(name);
30 endfunction: new
31 
32 reg_block_B model;
33 
34 virtual task body();
35 uvm_status_e status;
36 uvm_reg_data_t data;
37 int n;
38 
39 // Initialize R with a random value then check against mirror
40 data[7:0] = $urandom();
41 model.R.write(status, data, .parent(this));
42 model.R.mirror(status, UVM_CHECK, .parent(this));
43 
44 // Perform a random number of INC operations
45 n = ($urandom() % 7) + 3;
46 `uvm_info("blk_R_test_seq", $sformatf("Incrementing R %0d times...", n), UVM_NONE);
47 repeat (n) begin
48 model.CTL.write(status, reg_fld_B_CTL_CTL::INC, .parent(this));
49 data++;
50 void'(model.R.predict(data));
51 end
52 // Check the final value
53 model.R.mirror(status, UVM_CHECK, .parent(this));
54 
55 // Perform a random number of DEC operations
56 n = ($urandom() % 8) + 2;
57 `uvm_info("blk_R_test_seq", $sformatf("Decrementing R %0d times...", n), UVM_NONE);
58 repeat (n) begin
59 model.CTL.write(status, reg_fld_B_CTL_CTL::DEC, .parent(this));
60 data--;
61 void'(model.R.predict(data));
62 end
63 // Check the final value
64 model.R.mirror(status, UVM_CHECK, .parent(this));
65 
66 // Reset the register and check
67 model.CTL.write(status, reg_fld_B_CTL_CTL::CLR, .parent(this));
68 void'(model.R.predict(0));
69 model.R.mirror(status, UVM_CHECK, .parent(this));
70 endtask
71
72endclass
