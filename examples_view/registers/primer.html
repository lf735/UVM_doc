RAL Primer & Bus Adapter

Comprehensive introductory guide to the UVM Register Abstraction Layer: creating a register model, integrating a register adapter (reg2bus/bus2reg), and running register sequences.

Key UVM APIs: uvm_reg_block uvm_reg uvm_reg_field uvm_reg_adapter uvm_reg_predictor

Simulation Flow Stepper

Interactive Execution Progression
1High-Level Register Write
2Adapter Translation (reg2bus)
3Physical Bus Execution
4Monitor Observation & Predictor
5Mirror Update & Verification

1. High-Level Register Write

Target: user_sequence

The user sequence executes a high-level write on the RAL model (`reg_model.regA.write()`). The RAL model validates fields and updates its desired value.

Expected Log Output: UVM_INFO @ 10 ns: seq [RAL] reg_model.regA.write(status, 32'hA5A5_00FF)

Testbench Architecture Topology

uvm_reg_block reg_model
High-level RAL model
uvm_reg_adapter adapter
reg2bus / bus2reg
agent bus_agent
Drives physical APB bus

26module slave(apb_if apb,
27 input bit rst);
28 
29reg [31:0] pr_data;
30assign apb.prdata = (apb.psel && apb.penable && !apb.pwrite) ? pr_data : 'z;
31 
32typedef struct {
33 reg [63:0] SRC;
34 reg [63:0] DST;
35} socket_t;
36 
37reg [ 7:0] INDEX;
38socket_t SESSION[256];
39reg [63:0] DST[256];
40reg [31:0] TABLES[256];
41reg [31:0] DMA[1024];
42 
43always @ (posedge apb.pclk)
44 begin
45 if (rst) begin
46 INDEX <= 'h00;
47 foreach (TABLES[i]) begin
48 TABLES[i] <= 32'h0000_0000;
49 end
50 pr_data <= 32'h0;
51 end
52 else begin
53 
54 // Wait for a SETUP+READ or ENABLE+WRITE cycle
55 if (apb.psel == 1'b1 && apb.penable == apb.pwrite) begin
56 pr_data <= 32'h0;
57 if (apb.pwrite) begin
58 casex (apb.paddr)
59 16'h0020: INDEX <= apb.pwdata;
60 16'h0024: begin
61 TABLES[INDEX] <= apb.pwdata;
62 end
63 16'h1XX0: SESSION[apb.paddr[11:2]].SRC[63:32] <= apb.pwdata;
64 16'h1XX4: SESSION[apb.paddr[11:2]].SRC[32: 0] <= apb.pwdata;
65 16'h1XX8: SESSION[apb.paddr[11:2]].DST[63:32] <= apb.pwdata;
66 16'h1XXC: SESSION[apb.paddr[11:2]].DST[32: 0] <= apb.pwdata;
67 16'h2XXX: DMA[apb.paddr[11:2]] <= apb.pwdata;
68 endcase
69 end
70 else begin
71 casex (apb.paddr)
72 16'h0000: pr_data <= {4'h0, 10'h176, 8'h5A, 8'h03};
73 16'h0020: pr_data <= {24'h0000, INDEX};
74 16'h0024: pr_data <= TABLES[INDEX];
75 16'h1XX0: pr_data <= SESSION[apb.paddr[11:2]].SRC[63:32];
76 16'h1XX4: pr_data <= SESSION[apb.paddr[11:2]].SRC[32: 0];
77 16'h1XX8: pr_data <= SESSION[apb.paddr[11:2]].DST[63:32];
78 16'h1XXC: pr_data <= SESSION[apb.paddr[11:2]].DST[32: 0];
79 16'h2XXX: pr_data <= DMA[apb.paddr[11:2]];
80 endcase
81 end
82 end
83 end
84end
85 
86endmodule

167class reg_block_slave extends uvm_reg_block;

169 reg_slave_ID ID;

173 rand reg_slave_SESSION SESSION[256];

185 endfunction

187 virtual function void build();
188 
189 // create
190 ID = reg_slave_ID::type_id::create("ID");
191 INDEX = reg_slave_INDEX::type_id::create("INDEX");
192 DATA = reg_slave_DATA::type_id::create("DATA");
193 foreach (SESSION[i])
194 SESSION[i] = reg_slave_SESSION::type_id::create($sformatf("SESSION[%0d]",i));
195 foreach (TABLES[i])
196 TABLES[i] = reg_slave_TABLES::type_id::create($sformatf("TABLES[%0d]",i));
197 DMA_RAM = mem_slave_DMA_RAM::type_id::create("DMA_RAM");
198 
199 // configure
200 ID.configure(this,null,"ID");
201 ID.build();
202 INDEX.configure(this,null,"INDEX");
203 INDEX.build();
204 foreach (SESSION[i]) begin
205 SESSION[i].configure(this,null,$sformatf("SESSION[%0d]",i));
206 SESSION[i].build();
207 end
208 foreach (TABLES[i]) begin
209 TABLES[i].configure(this,null,$sformatf("TABLES[%0d]",i));
210 TABLES[i].build();
211 end
212 // the SV LRM IEEE2009 is not clear if an array of class handles can be assigned to another array if the element types
213 // are assignment compatible OR if the LRM states the element types have to be 'equal' (the LRM states equal types)
214 // IUS requires per LRM 'equivalent' element types and does not accept assignment compatible types
215`ifdef INCA
216 begin
217 uvm_reg r[256];
218 foreach(TABLES[i])
219 r[i]=TABLES[i];
220 
221 DATA.configure(INDEX, r, this, null);
222 end
223`else
224 DATA.configure(INDEX, TABLES, this, null);
225`endif
226
227 DATA.build();
228
229 DMA_RAM.configure(this,"");
230 
231 // define default map
232 default_map = create_map("default_map", 'h0, 4, UVM_LITTLE_ENDIAN);
233 default_map.add_reg(ID, 'h0, "RW");
234 default_map.add_reg(INDEX, 'h20, "RW");
235 default_map.add_reg(DATA, 'h24, "RW");
236 foreach (SESSION[i])
237 SESSION[i].map(default_map, 'h1000 + 16 * i);
238 
239 default_map.add_mem(DMA_RAM, 'h2000, "RW");
240
241 // field handle aliases
242 REVISION_ID = ID.REVISION_ID;
243 CHIP_ID = ID.CHIP_ID;
244 PRODUCT_ID = ID.PRODUCT_ID;
245 endfunction
