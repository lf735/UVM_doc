Broadcast Register Writes

Models broadcast register writes where a single write transaction to a broadcast address updates multiple distinct register instances simultaneously.

Key UVM APIs: uvm_reg_block write predict

Testbench Architecture Topology

write-only broadcast_reg
Updates reg1, reg2, reg3

24module blk(apb_if apb, input bit rst);
25 
26reg [2:0] mode;
27reg [7:0] rate;
28 
29reg [31:0] pr_data;
30 
31assign apb.prdata = (apb.psel && apb.penable) ? pr_data : 'z;
32 
33always @ (posedge apb.pclk)
34 begin
35 if (rst) begin
36 mode <= 0;
37 rate <= 0;
38 end
39 else begin
40 // Wait for a SETUP+READ or ENABLE+WRITE cycle
41 if (apb.psel == 1'b1 && apb.penable == apb.pwrite) begin
42 pr_data = 32'h0;
43 case (apb.paddr[7:0])
44 8'h00: if (apb.pwrite) mode <= apb.pwdata;
45 else pr_data = mode;
46 8'h04: if (apb.pwrite) rate <= apb.pwdata;
47 else pr_data = rate;
48 endcase
49 end
50 end
51end
52 
53endmodule

57module soc(apb_if apb, input bit rst);
58 
59apb_if apb0(apb.pclk);
60apb_if apb1(apb.pclk);
61apb_if apb2(apb.pclk);
62 
63blk blk0(apb0, rst);
64blk blk1(apb1, rst);
65blk blk2(apb2, rst);
66 
67assign apb0.paddr = apb.paddr;
68assign apb0.psel = apb.paddr[31:8] == 24'h000000 ||
69 apb.paddr[31:8] == 24'h030000;
70assign apb0.penable = apb.penable;
71assign apb0.pwrite = apb.pwrite;
72assign apb0.pwdata = apb.pwdata;
73assign apb.prdata = apb0.prdata;
74 
75assign apb1.paddr = apb.paddr;
76assign apb1.psel = apb.paddr[31:8] == 24'h010000 ||
77 apb.paddr[31:8] == 24'h030000;
78assign apb1.penable = apb.penable;
79assign apb1.pwrite = apb.pwrite;
80assign apb1.pwdata = apb.pwdata;
81assign apb.prdata = apb1.prdata;
82 
83assign apb2.paddr = apb.paddr;
84assign apb2.psel = apb.paddr[31:8] == 24'h020000 ||
85 apb.paddr[31:8] == 24'h030000;
86assign apb2.penable = apb.penable;
87assign apb2.pwrite = apb.pwrite;
88assign apb2.pwdata = apb.pwdata;
89assign apb.prdata = apb2.prdata;
90 
91endmodule
