Register Sequence Convenience API

Demonstrates the write_reg and read_reg convenience methods in user sequences, providing automatic parent sequencing without manual sequence context passing.

Key UVM APIs: write_reg read_reg uvm_reg_sequence

Testbench Architecture Topology

seq sequence
Calls write_reg / read_reg

23module blk_dut #(int BASE_ADDR='h0) (apb_if apb,
24 input bit rst);
25 
26reg [7:0] R;
27 
28reg [31:0] pr_data;
29 
30wire in_range;
31 
32wire [31:0] pr_addr;
33 
34assign in_range = (apb.paddr - BASE_ADDR) < 'h100;
35assign pr_addr = apb.paddr - BASE_ADDR;
36 
37assign apb.prdata = (apb.psel && apb.penable && !apb.pwrite && in_range) ? pr_data : 'z;
38 
39 
40always @ (posedge apb.pclk)
41 begin
42 if (rst) begin
43 R <= 'h00;
44 pr_data <= 32'h0;
45 end
46 else begin
47 
48 // Wait for a SETUP+READ or ENABLE+WRITE cycle
49 if (apb.psel == 1'b1 && apb.penable == apb.pwrite) begin
50 pr_data <= 32'h0;
51 if (apb.pwrite) begin
52 casex (pr_addr)
53 32'h00000000:
54 R <= apb.pwdata[7:0];
55 32'h00000001:
56 casez (apb.pwdata[1:0])
57 2'b01: R++;
58 2'b10: R--;
59 2'b11: R <= 0;
60 endcase
61 endcase
62 end
63 else begin
64 casex (pr_addr)
65 32'h00000000: pr_data <= {24'h0, R};
66 default: pr_data <= 32'h0;
67 endcase
68 #1;
69 end
70 #0;
71 end
72 end
73end
74 
75endmodule

25// This example uses the reg/mem write/read convenience API in the
26// base sequence class. The convenience methods provide the ~parent~
27// argument for you when calling the corresponding method in the
28// register or memory.
29 
30class blk_R_test_seq extends uvm_reg_sequence;
31 
32 `uvm_object_utils(blk_R_test_seq)
33 
34 function new(string name = "blk_R_test_seq");
35 super.new(name);
36 endfunction: new
37 
38 reg_block_B model;
39 
40 virtual task body();
41 uvm_status_e status;
42 uvm_reg_data_t data, rd_data;
43 int n;
44 
45 // Initialize R with a random value then check against mirror
46 data[7:0] = $urandom();
47 
48 write_reg(model.R, status, data);
49 read_reg (model.R, status, rd_data);
50 
51 if (data != rd_data)
52 `uvm_error("MISCOMPARE","Unexpected value on read")
53 
54 model.R.set(23);
55 update_reg(model.R, status);
56 mirror_reg(model.R, status, UVM_CHECK);
57 
58 data[7:0] = $urandom();
59 
60 poke_reg(model.R, status, data);
61 peek_reg(model.R, status, rd_data);
62 
63 if (data != rd_data)
64 `uvm_error("MISCOMPARE","Unexpected value on peek")
65 
66 // Perform a random number of INC operations
67 n = ($urandom() % 7) + 3;
68 `uvm_info("blk_R_test_seq", $sformatf("Incrementing R %0d times...", n), UVM_NONE);
69 repeat (n) begin
70 write_reg(model.CTL, status, reg_fld_B_CTL_CTL::INC);
71 data++;
72 void'(model.R.predict(data));
73 end
74 // Check the final value
75 mirror_reg(model.R, status, UVM_CHECK);
76 
77 // Perform a random number of DEC operations
78 n = ($urandom() % 8) + 2;
79 `uvm_info("blk_R_test_seq", $sformatf("Decrementing R %0d times...", n), UVM_NONE);
80 repeat (n) begin
81 write_reg(model.CTL, status, reg_fld_B_CTL_CTL::DEC);
82 data--;
83 void'(model.R.predict(data));
84 end
85 // Check the final value
86 mirror_reg(model.R, status, UVM_CHECK);
87 
88 // Reset the register and check
89 write_reg(model.CTL, status, reg_fld_B_CTL_CTL::CLR);
90 void'(model.R.predict(0));
91 mirror_reg(model.R, status, UVM_CHECK);
92 endtask
93
94endclass
