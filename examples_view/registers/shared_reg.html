Multi-Domain Shared Registers

Models registers accessible from multiple bus masters or clock domains with separate bus maps and synchronization semantics.

Key UVM APIs: create_map add_submap default_map

Testbench Architecture Topology

reg shared_reg
Mapped in both CPU_map and DMA_map

23class blk_AXW_test_seq extends uvm_reg_sequence;
24 
25 `uvm_object_utils(blk_AXW_test_seq)
26 
27 reg_block_B model;
28 
29 function new(string name = "blk_AXW_test_seq");
30 super.new(name);
31 endfunction: new
32 
33 virtual task body();
34 uvm_status_e status;
35 uvm_reg_data_t data;
36 int n;
37 
38 // Write all three registers at once. W will be written concurrently with
39 // A or X because W will go across the WSH bus, whereas A and X both go
40 // across the APB bus and must therefore execute sequentially. It the
41 // responsibility of the DUT to handle concurrent accesses across
42 // two different interfaces.
43 fork
44 model.A.write(status, 'h33, .parent(this));
45 model.X.write(status, 'h44, .parent(this));
46 model.W.write(status, 'hcc, .parent(this));
47 join
48 
49 // Write A to random value via default map (APB), then check against mirror
50 data[7:0] = $urandom();
51 model.A.write(status, data, .parent(this));
52 model.A.mirror(status, UVM_CHECK, .parent(this));
53 
54 // Write ~A to X via APB interface, then check via WSH interface
55 model.X.write(status, ~model.A.get(), .map(model.APB), .parent(this));
56 model.X.mirror(status, UVM_CHECK, .map(model.WSH), .parent(this));
57 
58 // Write ~X to W via default map (APB), then check
59 model.W.write(status, ~model.X.get(), .parent(this));
60 model.W.mirror(status, UVM_CHECK, .parent(this));
61 
62 // W should now be equal to A
63 if (model.W.get() !== data) begin
64 `uvm_error("test", $sformatf("W == 'h%h != 'h%h", model.X.get(), data));
65 end
66 endtask
67
68endclass
