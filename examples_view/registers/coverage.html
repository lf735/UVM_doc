Register Functional Coverage

Integrates SystemVerilog covergroups directly within register and field definitions to track address coverage, field value coverage, and access type coverage.

Key UVM APIs: uvm_reg_block uvm_reg build_coverage sample

Testbench Architecture Topology

RAL reg_model
Embedded covergroups sampled on read/write

22//
23// This example demonstrates how to include a coverage model
24// in a register model.
25//
26// Three coverage models are included:
27// - Register level coverage
28// - Address-leel coverage
29// - Field value coverage
30//
31// This example demonstrates *how* to integrate a coverage model
32// in a UVM Register model. The details of the coverage model itself
33// are generator-specific and outside the scope of UVM.
34//

36class reg_R extends uvm_reg;
37 rand uvm_reg_field F1;
38 rand uvm_reg_field F2;
39 
40 local uvm_reg_data_t m_current;
41 local uvm_reg_data_t m_data;
42 local uvm_reg_data_t m_be;
43 local bit m_is_read;
44 
45 covergroup cg_bits;
46 wF1_0: coverpoint {m_current[0],m_data[0]} iff (!m_is_read && m_be[0]);
47 wF1_1: coverpoint {m_current[1],m_data[1]} iff (!m_is_read && m_be[0]);
48 wF1_2: coverpoint {m_current[2],m_data[2]} iff (!m_is_read && m_be[0]);
49 wF1_3: coverpoint {m_current[3],m_data[3]} iff (!m_is_read && m_be[0]);
50 wF2_0: coverpoint {m_current[4],m_data[4]} iff (!m_is_read && m_be[0]);
51 wF2_1: coverpoint {m_current[5],m_data[5]} iff (!m_is_read && m_be[0]);
52 wF2_2: coverpoint {m_current[6],m_data[6]} iff (!m_is_read && m_be[0]);
53 wF2_3: coverpoint {m_current[7],m_data[7]} iff (!m_is_read && m_be[0]);
54 endgroup
55
56 covergroup cg_vals;
57 F1: coverpoint F1.value[3:0];
58 F2: coverpoint F2.value[3:0];
59 F1F2: cross F1, F2;
60 endgroup

138class block_B extends uvm_reg_block;
139 rand reg_R Ra;
140 rand reg_R Rb;
141 
142 mem_M M;
143 
144 local uvm_reg_addr_t m_offset;
145 
146 covergroup cg_addr;
147 Ra: coverpoint m_offset
148 {
149 bins hit = {'h0000};
150 }
151 Rb: coverpoint m_offset
152 {
153 bins hit = {'h0100};
154 }
155 M: coverpoint m_offset
156 {
157 bins MIN = {'h2000};
158 bins MAX = {'h23FF};
159 }
160 endgroup
161 
162 covergroup cg_vals;
163 Ra: coverpoint Ra.F1.value[3:0];
164 Rb: coverpoint Rb.F1.value[3:0];
165 RaRb: cross Ra, Rb;
166 endgroup
167
168
169 function new(string name = "B");
170 super.new(name, build_coverage(UVM_CVR_ADDR_MAP+UVM_CVR_FIELD_VALS));
171 if (has_coverage(UVM_CVR_ADDR_MAP))
172 cg_addr = new();
173 if (has_coverage(UVM_CVR_FIELD_VALS))
174 cg_vals = new();
175 endfunction: new
176
177 virtual function void build();
178 
179 default_map = create_map("", 0, 4, UVM_BIG_ENDIAN, 0);
180 
181 Ra = reg_R::type_id::create("Ra",,get_full_name());
182 Ra.configure(this, null);
183 Ra.build();
184 
185 Rb = reg_R::type_id::create("Rb",,get_full_name());
186 Rb.configure(this, null);
187 Rb.build();
188 
189 M = mem_M::type_id::create("M",,get_full_name());
190 M.configure(this);
191 
192 default_map.add_reg(Ra, 'h0000, "RW");
193 default_map.add_reg(Rb, 'h0100, "RW");
194 default_map.add_mem(M, 'h2000, "RW");
195 
196 endfunction : build

26class dut;
27 static bit [3:0] Ra_F1 = 0;
28 static bit [3:0] Ra_F2 = 0;
29 static bit [3:0] Rb_F1 = 0;
30 static bit [3:0] Rb_F2 = 0;
31 static bit [7:0] M[1024];
32 
33 static task rw(reg_rw rw);
34 casez (rw.addr)
35 'h0000: // Ra
36 if (rw.read) rw.data = {16'h00, Ra_F2, Ra_F1};
37 else begin
38 if (rw.byte_en[0]) Ra_F1 = rw.data[ 7:0];
39 if (rw.byte_en[0]) Ra_F2 = rw.data[15:8];
40 end
41 'h0100: // Rb
42 if (rw.read) rw.data = {16'h00, Rb_F2, Rb_F1};
43 else begin
44 if (rw.byte_en[0]) Rb_F1 = rw.data[ 7:0];
45 if (rw.byte_en[0]) Rb_F2 = rw.data[15:8];
46 end
47 'b001000??_????????: // M
48 if (rw.read) rw.data = {24'h00, M[rw.addr[9:0]]};
49 else begin
50 if (rw.byte_en[0]) M[rw.addr[9:0]] = rw.data[ 7:0];
51 end
52 endcase
53 #1;
54 // $write("%s\n", rw.convert2string());
55 endtask
