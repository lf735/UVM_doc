User-Defined Register Callbacks

Extends standard register access behaviors with custom pre_read, post_read, pre_write, and post_write user callbacks for side-effects.

Key UVM APIs: uvm_reg_cbs pre_write post_write pre_read post_read

Testbench Architecture Topology

reg reg_ctrl
Has user_callback attached

28//
29// The user_acp_reg has a user-defined behavior
30//
31// It increments by 1 after every write
32//
33class user_acp_incr_on_write_cbs extends uvm_reg_cbs;
34 
35 virtual function void post_predict(input uvm_reg_field fld,
36 input uvm_reg_data_t previous,
37 inout uvm_reg_data_t value,
38 input uvm_predict_e kind,
39 input uvm_path_e path,
40 input uvm_reg_map map);
41 if (kind != UVM_PREDICT_WRITE) return;
42 if (path != UVM_FRONTDOOR) return;
43 
44 value = previous + 1;
45 endfunction
46
47endclass

50class user_acp_reg extends uvm_reg;
51 
52 local uvm_reg_field value;
53 
54 `uvm_object_utils(user_acp_reg)
55 
56 function new(string name = "user_acp_reg");
57 super.new(name,16,UVM_NO_COVERAGE);
58 endfunction: new
59 
60 virtual function void build();
61 value = uvm_reg_field::type_id::create("value",,get_full_name());
62 value.configure(this, 16, 0, "RW", 0, 16'h0000, 1, 0, 0);
63 value.set_compare(UVM_NO_CHECK);
64 
65 uvm_resource_db#(bit)::set({"REG::",get_full_name()},
66 "NO_REG_BIT_BASH_TEST", 1);
67 uvm_resource_db#(bit)::set({"REG::",get_full_name()},
68 "NO_REG_ACCESS_TEST", 1);
69 
70 begin
71 user_acp_incr_on_write_cbs cb = new;
72 uvm_reg_field_cb::add(value, cb);
73 end
74 endfunction: build
75 
76 
77 virtual task pre_write(uvm_reg_item rw);
78 uvm_reg_data_t m_data;
79 uvm_reg rg;
80 
81 assert($cast(rg,rw.element));
82 
83 // Predict the value that will be in the register
84 m_data = rg.get() + 1;
85
86 // If a backdoor write is used, replace the value written
87 // with the incremented value to emulate the front-door
88 if (rw.path == UVM_BACKDOOR)
89 rw.value[0] = m_data;
90 endtask: pre_write
91 
92endclass : user_acp_reg

26class dut_rw;
27 static task rw(reg_rw rw);
28 casez (rw.addr)
29 'h0000: // acp (MSB)
30 if (rw.read) rw.data = dut.acp[15:8];
31 'h0001: // acp (LSB)
32 if (rw.read) rw.data = dut.acp[7:0];
33 else dut.acp++;
34 endcase
35 #1;
36 // $write("%s\n", rw.convert2string());
37 endtask
38endclass

42 virtual task run_phase(uvm_phase phase);
43 tb_env env;
44 uvm_status_e status;
45 uvm_reg_data_t data;
46 
47 phase.raise_objection(this);
48
49 if (!$cast(env, uvm_top.find("env")) || env == null) begin
50 `uvm_fatal("test", "Cannot find tb_env");
51 end
52 
53 env.regmodel.reset();
54 
55 begin
56 uvm_reg_sequence seq;
57 
58 seq = uvm_reg_hw_reset_seq::type_id::create("seq");
59 seq.model = env.regmodel;
60 seq.start(null);
61 seq.wait_for_sequence_state(UVM_FINISHED);
62 end
63 
64 `uvm_info("Test", "Performing 257 writes...", UVM_LOW)
65 
66 repeat (257) begin
67 env.regmodel.user_acp.write(status, $random());
68 end
69 env.regmodel.user_acp.mirror(status, UVM_CHECK);
70 
71 `uvm_info("Test", "Resetting DUT...", UVM_LOW)
72 dut.reset();
73 env.regmodel.reset();
74 env.regmodel.user_acp.mirror(status, UVM_CHECK);
75
