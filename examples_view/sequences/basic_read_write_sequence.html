Basic Read/Write Sequence

Demonstrates standard transaction item generation, sequencer arbitration, and the driver-sequencer handshake via start_item() and finish_item().

Key UVM APIs: uvm_sequence uvm_sequencer uvm_sequence_item start_item finish_item

Testbench Architecture Topology

seq sequence
Generates items
sqr sequencer
Arbitration
drv driver
Executes pins

37/*
38
39Remark 1:
40
41- this example shows two alternative approaches to handle local properties -
42
43When the define USE_FIELD_MACROS is set, the code is utilizing the uvm_field_xxx macros
44to mark transaction fields as "uvm fields". Doing so automatically infers code which handles
45compare,print,copy,record,pack,unpack without any additional implementation from the user side.
46This is typically shorter to write and easier to maintain on the user side. On the other hand a generic
47implementation might not be as fast and custom tailored.
48
49The two styles can be mixed for most of the uvm_field functionality (manually handling few fields and
50use a default implementation for some others). Only the component level auto configuration for fields is for
51`uvm_field_* annotated members only. it means that only when the uvm_field_* macros are used these fields will be
52set to their configured values automatically.
53
54Remark 2:
55
56All classes in this example utilize the uvm_(object|component)[_param]_utils macro. Althrough its not mandatory
57for UVM it is good practice since this macro will register the class in the UVM factory automatically. See the user
58guide for more information regarding uvm factory use models
59
60*/

79`ifdef USE_FIELD_MACROS

84 `uvm_field_enum(bus_op_t,op,UVM_DEFAULT)

91 virtual function void do_copy (uvm_object rhs);
92 bus_trans rhs_;
93 if(!$cast(rhs_, rhs))
94 `uvm_error("do_copy", "cast failed, check type compatability")
95 
96 super.do_copy(rhs);
97 
98 addr = rhs_.addr;
99 data = rhs_.data;
100 op = rhs_.op;
101 endfunction
102 
103 virtual function bit do_compare(uvm_object rhs,uvm_comparer comparer);
104 bus_trans rhs_;
105 if(!$cast(rhs_, rhs))
106 `uvm_fatal("do_compare", "cast failed, check type compatability")
107 
108 return ((op == rhs_.op) && (addr == rhs_.addr) && (data == rhs_.data));
109 endfunction
110 
111 virtual function string convert2string();
112 return $sformatf("op %s: addr=%03x, data=%02x", op.name(), addr, data);
113 endfunction

168 class my_driver #(type REQ = uvm_sequence_item,
169 type RSP = uvm_sequence_item) extends uvm_driver #(REQ, RSP);
170 
171 `uvm_component_param_utils(my_driver#(REQ,RSP))
172 
173 local int data_array[511:0];
174 
175 function new(string name, uvm_component parent);
176 super.new(name, parent);
177 endfunction

179 virtual task run_phase(uvm_phase phase);
180 REQ req;
181 RSP rsp;
182 
183 forever begin
184 seq_item_port.get(req);
185 rsp = new();
186 rsp.set_id_info(req);
187 
188 // Actually do the read or write here
189 if (req.op == BUS_READ) begin
190 rsp.addr = req.addr[8:0];
191 rsp.data = data_array[rsp.addr];
192 `uvm_info("sending",`toSTRING(rsp),UVM_MEDIUM)
193 end else begin
194 data_array[req.addr[8:0]] = req.data;
195 `uvm_info("sending",`toSTRING(req),UVM_MEDIUM)
196 end
197 seq_item_port.put(rsp);
198 end
199 endtask

204 class sequenceA #(type REQ = uvm_sequence_item,
205 type RSP = uvm_sequence_item) extends uvm_sequence #(REQ, RSP);
206 
207 `uvm_object_param_utils(sequenceA#(REQ,RSP))
208 
209 local static integer g_my_id = 1;
210 local integer my_id;
211 
212 function new(string name="");
213 super.new(name);
214 my_id = g_my_id++;
215 endfunction
216 
217 virtual task body();
218 REQ req;
219 RSP rsp;
220 
221 `uvm_info("sequenceA", "Starting sequence", UVM_MEDIUM)
222 
223 for(int unsigned i = 0; i < `NUM_LOOPS; i++) begin
224 `uvm_create(req)
225 
226 req.addr = (my_id * `NUM_LOOPS) + i;
227 req.data = my_id + i + 55;
228 req.op = BUS_WRITE;
229 
230 `uvm_send(req)
231 get_response(rsp);
232 
233 `uvm_create(req)
234 req.addr = (my_id * `NUM_LOOPS) + i;
235 req.data = 0;
236 req.op = BUS_READ;
237 
238 `uvm_send(req)
239 get_response(rsp);
240 
241 if (rsp.data != (my_id + i + 55)) begin
242 `uvm_error("SequenceA", $sformatf("Error, addr: %0d, expected data: %0d, actual data: %0d",
243 req.addr, req.data, rsp.data))
244 end
245 end
246 `uvm_info("sequenceA", "Finishing sequence", UVM_MEDIUM)
247 endtask // body
248 
249 endclass

251 class env extends uvm_env;
252 `uvm_component_utils(env)
253 local uvm_sequencer #(bus_req, bus_rsp) sqr;
254 local my_driver #(bus_req, bus_rsp) drv ;
255 
256 function new(string name, uvm_component parent);
257 super.new(name, parent);
258 endfunction
259 
260 virtual function void build_phase(uvm_phase phase);
261 super.build_phase(phase);
262 sqr = new("sequence_controller", this);
263 
264 // create and connect driver
265 drv = new("my_driver", this);
266 drv.seq_item_port.connect(sqr.seq_item_export);
267 endfunction
268 
269 virtual task run_phase(uvm_phase phase);
270 phase.raise_objection(this);
271 for (int i = 0; i < `NUM_SEQS; i++) begin
272 fork begin
273 sequenceA #(bus_req, bus_rsp) the_sequence= new("sequence");
274 the_sequence.start(sqr, null);
275 end
276 join_none
277 end
278 wait fork;
279 phase.drop_objection(this);
280 endtask
