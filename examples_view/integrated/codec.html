28module dut(output bit tx,
29 input bit rx,
30 input bit sclk,
31 apb_if apb,
32 output bit intr,
33 input bit clk,
34 input bit rst);
35 
36import uvm_pkg::*;
37uvm_report_object rpt;
38initial rpt = new($sformatf("%m"));
39 
40reg [31:0] pr_data;
41assign apb.prdata = (apb.psel && apb.penable && !apb.pwrite) ? pr_data : 'z;
42 
43reg TxEn;
44reg RxEn;
45 
46reg [ 7:0] TxFIFO[$];
47const int TxDepth = 32; // Depth of Tx FIFO
48reg [ 4:0] TxLWM; // Low Water Mark on TxFIFO
49 
50reg [ 7:0] RxFIFO[$];
51const int RxDepth = 32; // Depth of Rx FIFO
52reg [ 4:0] RxHWM; // High Water Mark on RxFIFO
53 
54const bit [7:0] SYNC = 8'hB2;
55const bit [7:0] ESC = 8'hE7;
56const bit [7:0] IDLE = 8'h81;
57 
58// Status bits
59reg TxEmpty;
60reg TxLow;
61reg TxFull;
62reg RxEmpty;
63reg RxHigh;
64reg RxFull;
65reg aligned;
66reg SA;
67 
68wire [8:0] IntReq = {SA,
69 1'b0, RxFull, RxHigh, RxEmpty,
70 1'b0, TxFull, TxLow, TxEmpty};
71reg [8:0] IntMask;
72assign intr = |(IntReq & IntMask);
73 
74bit [7:0] TxByte;
75bit TxValid;
76 
77bit [7:0] RxByte;
78bit RxValid;
79 
80always @ (posedge clk)
81begin
82 if (rst) begin
83 TxEn = 0;
84 TxFIFO.delete();
85 TxLWM <= TxDepth * 0.25;
86
87 RxEn = 0;
88 aligned = 0;
89 RxFIFO.delete();
90 RxHWM <= RxDepth * 0.50;
91 
92 pr_data <= 32'h0;
93 TxValid = 0;
94 end
95 else begin
96
97 // Wait for a SETUP+READ or ENABLE+WRITE cycle
98 if (apb.psel == 1'b1 && apb.penable == apb.pwrite) begin
99 pr_data <= 32'h0;
100 if (apb.pwrite) begin
101 casex (apb.paddr)
102 16'h0000: if (apb.pwdata[8]) SA = 0;
103 16'h0004: IntMask <= apb.pwdata;
104 16'h0010: TxEn <= apb.pwdata;
105 16'h0014: TxLWM <= apb.pwdata;
106 16'h0020: RxEn <= apb.pwdata;
107 16'h0024: RxHWM <= apb.pwdata;
108 16'h0100: if (TxFIFO.size() < TxDepth) TxFIFO.push_back(apb.pwdata);
109 endcase
110 end
111 else begin
112 casex (apb.paddr)
113 16'h0000: pr_data <= IntReq;
114 16'h0004: pr_data <= IntMask;
115 16'h0010: pr_data <= TxEn;
116 16'h0014: pr_data <= TxLWM;
117 16'h0020: pr_data <= {aligned, RxEn};
118 16'h0024: pr_data <= RxHWM;
119 16'h0100: if (RxFIFO.size() > 0) begin
120 pr_data <= RxFIFO[0];
121 void'(RxFIFO.pop_front());
122 end
123 endcase
124 end
125 end
126 end
127 
128 //
129 // Tx
130 //
131 if (!TxValid && TxFIFO.size() > 0) begin
132 TxByte = TxFIFO[0];
133 TxValid = 1;
134 void'(TxFIFO.pop_front());
135 end
136 TxEmpty <= TxFIFO.size() == 0;
137 TxLow <= TxFIFO.size() <= TxLWM;
138 TxFull <= TxFIFO.size() >= TxDepth;
139 
140 //
141 // Rx
142 //
143 if (RxValid) begin
144 if (RxFIFO.size() < RxDepth) RxFIFO.push_back(RxByte);
145 RxValid = 0;
146 end
147 RxEmpty <= RxFIFO.size() == 0;
148 RxHigh <= RxFIFO.size() >= RxHWM;
149 RxFull <= RxFIFO.size() >= TxDepth;
150end
151 
152 
153//
154// Transmit bytes, with SYNC symbols every 7 bytes
155//
156always begin: TX
157 wait (rst != 1 && TxEn);
158 
159 if (TxValid) begin
160 bit [7:0] symbol;
161
162 symbol = TxByte;
163 TxValid = 0;
164
165 if (symbol == IDLE || symbol == ESC) send(ESC);
166 send(symbol);
167 end
168 else send(IDLE);
169end
170 
171task automatic send(input bit [7:0] symbol);
172 static int tx_cnt = 0;
173 
174 if (tx_cnt == 0) begin
175 bit [7:0] sync = 8'hB2;
176 repeat (8) begin
177 tx = sync[7];
178 @(negedge sclk);
179 sync = sync << 1;
180 end
181 end
182 repeat (8) begin
183 tx = symbol[7];
184 @(negedge sclk);
185 symbol = symbol << 1;
186 end
187 tx_cnt = (tx_cnt + 1) % 6;
188endtask
189 
190 
191//
192// Detect symbol alignment and receive bytes once aligned
193//
194always begin: RX
195 bit [7:0] symbol;
196 bit escaped;
197
198 RxValid = 0;
199 symbol = 8'h00;
200 escaped = 0;
201 aligned = 1'b0;
202 
203 wait (rst != 1 && RxEn);
204 
205 `uvm_info_context("RX", "Rx Path Enabled...", UVM_MEDIUM, rpt)
206
207 // First, look for SYNC in the bit stream
208 `uvm_info_context("RX", "Looking for SYNC character...", UVM_MEDIUM, rpt)
209 while (symbol != SYNC) begin
210 @(posedge sclk);
211 symbol = {symbol[6:0], rx};
212 end
213 `uvm_info_context("RX", "Found SYNC character!", UVM_MEDIUM, rpt)
214 
215 // Next, look for SYNC every 7 bytes for 3 frames
216 repeat (3) begin
217 repeat (21 * 8) begin
218 @ (posedge sclk);
219 symbol = {symbol[6:0], rx};
220 end
221 if (symbol != SYNC) disable RX;
222 `uvm_info_context("RX", "Found SYNC character!", UVM_MEDIUM, rpt)
223 end
224 
225 `uvm_info_context("RX", "Symbol alignment acquired!", UVM_MEDIUM, rpt)
226
227 // We are in phase!
228 SA = 1'b1;
229 aligned = 1'b1;
230 escaped = 0;
231 
232 // Make sure we continue to see SYNC every 7 bytes
233 forever begin
234 repeat (6) begin
235 repeat (8) begin
236 @ (posedge sclk);
237 symbol = {symbol[6:0], rx};
238 end
239 if (escaped || (symbol != IDLE && symbol != ESC)) begin
240 RxValid = 1;
241 RxByte = symbol;
242 escaped = 0;
243 end
244 else if (symbol == ESC) begin
245 escaped = 1;
246 end
247 end
248 repeat (8) begin
249 @ (posedge sclk);
250 symbol = {symbol[6:0], rx};
251 end
252 if (symbol != SYNC) begin
253 `uvm_info_context("RX", "Symbol alignment lost!", UVM_MEDIUM, rpt)
254 SA = 1;
255 disable RX;
256 end
257 end
258end
259 
260//
261// Reset handler
262//
263always @(posedge clk)
264begin
265 if (rst == 1'b1) begin
266 disable RX;
267 disable TX;
268 end
269 else begin
270 if (!TxEn) disable TX;
271 if (!RxEn) disable RX;
272 end
273end

275endmodule

271. DUT SPECIFICATION
28 
29The DUT is a full-duplex parallel-to-serial codec.
30 
31Bytes written to the TxFIFO are transmitted MSB-first in order
32received. If there are no bytes to be transmitted, IDLE is
33transmitted.
34 
35Received bytes are added to the RxFIFO if it is not full. SYNC, IDLE
36and ESC characters are ignored.
37 
38The bit stream is identical in both direction. There is a SYNC
39character (0xB2) inserted every 7 bytes. IDLE characters (0x81)
40represent invalid data, unless escaped by a preceeding ESC character
41(0xE7). If an IDLE or ESC character must be transmitted as valid data,
42they must be preceeded by a ESC characted. The SYNC character, when
43found in a position other than the SYNC slot is a valid character.
44 
45For example, the phrase 0xADB2EF81E7FEED could be transmitted as:
46 
47 B2 81 81 81 81 AD B2 B2 81 EF 81 E7 81 E7 B2 E7 81 E7 FE ED 81 B2
48 SS .. .. .. .. vv vv SS .. vv .. EE vv EE SS vv .. EE vv vv .. SS
49 
50SS = SYNC
51EE = ESC
52.. = IDLE
53vv = Valid data
54 
55Bits are valid at the rising edge of the clock.
