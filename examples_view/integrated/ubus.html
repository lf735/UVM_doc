64 // memory_verify
65 protected function void memory_verify(input ubus_transfer trans);
66 int unsigned data, exp;
67 for (int i = 0; i < trans.size; i++) begin
68 // Check to see if entry in associative array for this address when read
69 // If so, check that transfer data matches associative array data.
70 if (m_mem_expected.exists(trans.addr + i)) begin
71 if (trans.read_write == READ) begin
72 data = trans.data[i];
73 `uvm_info(get_type_name(),
74 $sformatf("%s to existing address...Checking address : %0h with data : %0h",
75 trans.read_write.name(), trans.addr, data), UVM_LOW)
76 assert(m_mem_expected[trans.addr + i] == trans.data[i]) else begin
77 exp = m_mem_expected[trans.addr + i];
78 `uvm_error(get_type_name(),
79 $sformatf("Read data mismatch. Expected : %0h. Actual : %0h",
80 exp, data))
81 sbd_error = 1;
82 end
83 num_init_reads++;
84 end
85 if (trans.read_write == WRITE) begin
86 data = trans.data[i];
87 `uvm_info(get_type_name(),
88 $sformatf("%s to existing address...Updating address : %0h with data : %0h",
89 trans.read_write.name(), trans.addr + i, data), UVM_LOW)
90 m_mem_expected[trans.addr + i] = trans.data[i];
91 num_writes++;
92 end
93 end
94 // Check to see if entry in associative array for this address
95 // If not, update the location regardless if read or write.
96 else begin
97 data = trans.data[i];
98 `uvm_info(get_type_name(),
99 $sformatf("%s to empty address...Updating address : %0h with data : %0h",
100 trans.read_write.name(), trans.addr + i, data), UVM_LOW)
101 m_mem_expected[trans.addr + i] = trans.data[i];
102 if(trans.read_write == READ)
103 num_uninit_reads++;
104 else if (trans.read_write == WRITE)
105 num_writes++;
106 end
107 end
108 endfunction : memory_verify

232 // collect_arbitration_phase
233 task collect_arbitration_phase();
234 string tmpStr;
235 @(posedge vif.sig_clock iff (vif.sig_grant != 0));
236 status.bus_state = ARBI;
237 state_port.write(status);
238 void'(this.begin_tr(trans_collected));
239 // Check which grant is asserted to determine which master is performing
240 // the transfer on the bus.
241 for (int j = 0; j <= 15; j++) begin
242 if (vif.sig_grant[j] === 1) begin

246 end

250 // collect_address_phase
251 task collect_address_phase();
252 @(posedge vif.sig_clock);
253 trans_collected.addr = vif.sig_addr;
254 case (vif.sig_size)
255 2'b00 : trans_collected.size = 1;
256 2'b01 : trans_collected.size = 2;
257 2'b10 : trans_collected.size = 4;
258 2'b11 : trans_collected.size = 8;
259 endcase
260 trans_collected.data = new[trans_collected.size];
261 case ({vif.sig_read,vif.sig_write})
262 2'b00 : begin
263 trans_collected.read_write = NOP;
264 status.bus_state = NO_OP;
265 state_port.write(status);
266 end
267 2'b10 : begin
268 trans_collected.read_write = READ;
269 status.bus_state = ADDR_PH;
270 state_port.write(status);
271 end
272 2'b01 : begin
273 trans_collected.read_write = WRITE;
274 status.bus_state = ADDR_PH;
275 state_port.write(status);
276 end
277 2'b11 : begin
278 status.bus_state = ADDR_PH_ERROR;
279 state_port.write(status);
280 if (checks_enable)
281 `uvm_error(get_type_name(),
282 "Read and Write true at the same time")
283 end
284 endcase
285 endtask : collect_address_phase

287 // collect_data_phase

289 int i;
290 if (trans_collected.read_write != NOP) begin
291 check_which_slave();
292 for (i = 0; i < trans_collected.size; i++) begin
293 status.bus_state = DATA_PH;
294 state_port.write(status);
295 @(posedge vif.sig_clock iff vif.sig_wait === 0);
296 trans_collected.data[i] = vif.sig_data;
297 end
298 num_transactions++;
299 this.end_tr(trans_collected);
300 end
301 endtask : collect_data_phase

303 // check_which_slave
304 function void check_which_slave();
305 string slave_name;
306 bit slave_found;
307 slave_found = 1'b0;
308 if(slave_addr_map.first(slave_name))
309 do begin
310 if (slave_addr_map[slave_name].get_min_addr() <= trans_collected.addr
311 && trans_collected.addr <= slave_addr_map[slave_name].get_max_addr())
312 begin
313 trans_collected.slave = slave_name;
314 slave_found = 1'b1;
315 end
316 if (slave_found == 1'b1)
317 break;
318 end
319 while (slave_addr_map.next(slave_name));
320 assert(slave_found) else begin
321 `uvm_error(get_type_name(),
322 $sformatf("Master attempted a transfer at illegal address 16'h%0h",
323 trans_collected.addr))
324 end
325 endfunction : check_which_slave

60 // get_and_drive
61 virtual protected task get_and_drive();
62 @(negedge vif.sig_reset);
63 forever begin
64 @(posedge vif.sig_clock);
65 seq_item_port.get_next_item(req);
66 $cast(rsp, req.clone());
67 rsp.set_id_info(req);

74 // reset_signals
75 virtual protected task reset_signals();
76 forever begin
77 @(posedge vif.sig_reset);
78 vif.sig_request[master_id] <= 0;
79 vif.rw <= 'h0;
80 vif.sig_addr <= 'hz;
81 vif.sig_data_out <= 'hz;
82 vif.sig_size <= 'bz;
83 vif.sig_read <= 'bz;
84 vif.sig_write <= 'bz;
85 vif.sig_bip <= 'bz;
86 end
87 endtask : reset_signals

89 // drive_transfer
90 virtual protected task drive_transfer (ubus_transfer trans);
91 if (trans.transmit_delay > 0) begin
92 repeat(trans.transmit_delay) @(posedge vif.sig_clock);
93 end
94 arbitrate_for_bus();
95 drive_address_phase(trans);
96 drive_data_phase(trans);
97 endtask : drive_transfer
98 
99 // arbitrate_for_bus
100 virtual protected task arbitrate_for_bus();
101 vif.sig_request[master_id] <= 1;
102 @(posedge vif.sig_clock iff vif.sig_grant[master_id] === 1);
103 vif.sig_request[master_id] <= 0;
104 endtask : arbitrate_for_bus
105 
106 // drive_address_phase
107 virtual protected task drive_address_phase (ubus_transfer trans);
108 vif.sig_addr <= trans.addr;
109 drive_size(trans.size);
110 drive_read_write(trans.read_write);
111 @(posedge vif.sig_clock);
112 vif.sig_addr <= 32'bz;
113 vif.sig_size <= 2'bz;
114 vif.sig_read <= 1'bz;
115 vif.sig_write <= 1'bz;
116 endtask : drive_address_phase
117 
118 // drive_data_phase
119 virtual protected task drive_data_phase (ubus_transfer trans);
120 bit err;
121 for(int i = 0; i <= trans.size - 1; i ++) begin
122 if (i == (trans.size - 1))
123 vif.sig_bip <= 0;
124 else
125 vif.sig_bip <= 1;
126 case (trans.read_write)
127 READ : read_byte(trans.data[i], err);
128 WRITE : write_byte(trans.data[i], err);
129 endcase
130 end //for loop
131 vif.sig_data_out <= 8'bz;
132 vif.sig_bip <= 1'bz;
133 endtask : drive_data_phase
134 
135 // read_byte
136 virtual protected task read_byte (output bit [7:0] data, output bit error);
137 vif.rw <= 1'b0;
138 @(posedge vif.sig_clock iff vif.sig_wait === 0);
139 data = vif.sig_data;
140 endtask : read_byte
141 
142 // write_byte
143 virtual protected task write_byte (bit[7:0] data, output bit error);
144 vif.rw <= 1'b1;
145 vif.sig_data_out <= data;
146 @(posedge vif.sig_clock iff vif.sig_wait === 0);
147 vif.rw <= 'h0;
148 endtask : write_byte
149 
150 // drive_size
151 virtual protected task drive_size (int size);
152 case (size)
153 1: vif.sig_size <= 2'b00;
154 2: vif.sig_size <= 2'b01;
155 4: vif.sig_size <= 2'b10;
156 8: vif.sig_size <= 2'b11;
157 endcase
158 endtask : drive_size
159 
160 // drive_read_write
161 virtual protected task drive_read_write(ubus_read_write_enum rw);
162 case (rw)
163 NOP : begin vif.sig_read <= 0; vif.sig_write <= 0; end
164 READ : begin vif.sig_read <= 1; vif.sig_write <= 0; end
165 WRITE : begin vif.sig_read <= 0; vif.sig_write <= 1; end
166 endcase
167 endtask : drive_read_write

141 @(posedge vif.sig_clock);
142 trans_collected.addr = vif.sig_addr;
143 case (vif.sig_size)
144 2'b00 : trans_collected.size = 1;
145 2'b01 : trans_collected.size = 2;
146 2'b10 : trans_collected.size = 4;
147 2'b11 : trans_collected.size = 8;

150 case ({vif.sig_read,vif.sig_write})

162 @(posedge vif.sig_clock iff vif.sig_wait === 0);
163 trans_collected.data[i] = vif.sig_data;
164 end
165 this.end_tr(trans_collected);
166 endtask : collect_data_phase

41 // Basic arbiter, supports two masters, 0 has priority over 1
42 
43 always @(posedge ubus_clock or posedge ubus_reset) begin
44 if(ubus_reset) begin
45 ubus_start <= 1'b0;
46 st<=3'h0;
47 end
48 else
49 case(st)
50 0: begin //Begin out of Reset
51 ubus_start <= 1'b1;
52 st<=3'h3;
53 end
54 3: begin //Start state
55 ubus_start <= 1'b0;
56 if((ubus_gnt_master_0==0) && (ubus_gnt_master_1==0)) begin
57 st<=3'h4;
58 end
59 else
60 st<=3'h1;
61 end
62 4: begin // No-op state
63 ubus_start <= 1'b1;
64 st<=3'h3;
65 end
66 1: begin // Addr state
67 st<=3'h2;
68 ubus_start <= 1'b0;
69 end
70 2: begin // Data state
71 if((ubus_error==1) || ((ubus_bip==0) && (ubus_wait==0))) begin
72 st<=3'h3;
73 ubus_start <= 1'b1;
74 end
75 else begin
76 st<=3'h2;
77 ubus_start <= 1'b0;
78 end
79 end
80 endcase
81 end

84 if(ubus_reset == 1'b1) begin
85 ubus_gnt_master_0 <= 0;
86 ubus_gnt_master_1 <= 0;
87 end
