APB VIP Agent Library

Reusable AMBA APB (Advanced Peripheral Bus) UVC package: master agent, configurable sequencer, monitor, signal interface, and read/write transfer sequences.

Key UVM APIs: uvm_agent uvm_driver uvm_monitor uvm_sequencer uvm_sequence uvm_sequence_item

Source Specification & Design Notes

Test Intent & Implementation Walkthrough
Source: apb.sv

Advanced Peripheral Bus (APB) UVC Architecture

Complete AMBA APB (Advanced Peripheral Bus) verification component package:

  • APB Master Driver & Sequencer: Generates SETUP and ACCESS phase bus cycles
  • APB Monitor: Captures bus transactions from virtual interface pins
  • APB Agent & Config: Encapsulates driver, monitor, and sequencer with active/passive modes
  • Serves as the primary bus bridge for RAL register adapters throughout the UVM test suite

46 virtual function void build_phase(uvm_phase phase);
47 apb_agent agent;
48 if ($cast(agent, get_parent()) && agent != null) begin
49 sigs = agent.vif;
50 end
51 else begin
52 if (!uvm_config_db#(apb_vif)::get(this, "", "vif", sigs)) begin
53 `uvm_fatal("APB/DRV/NOVIF", "No virtual interface specified for this driver instance")
54 end

58 virtual task run_phase(uvm_phase phase);
59 super.run_phase(phase);
60 
61 this.sigs.mck.psel <= '0;
62 this.sigs.mck.penable <= '0;
63 
64 forever begin
65 apb_rw tr;
66 @ (this.sigs.mck);
67 
68 seq_item_port.get_next_item(tr);
69 
70 // TODO: QUESTA issue with hier ref to sequence via modport; need workaround?
71`ifdef VCS
72 if (!this.sigs.mck.at_posedge.triggered)
73`endif
74 
75`ifdef INCA
76 // FIXME if (!this.sigs.mck.at_posedge.triggered) // this is wrong and has to be reviewed
77`endif
78`ifdef QUESTA
79 if (!this.sigs.mck.triggered)
80`endif
81 @ (this.sigs.mck);
82
83 this.trans_received(tr);
84 `uvm_do_callbacks(apb_master,apb_master_cbs,trans_received(this,tr))
85
86 case (tr.kind)
87 apb_rw::READ: this.read(tr.addr, tr.data);
88 apb_rw::WRITE: this.write(tr.addr, tr.data);
89 endcase
90
91 this.trans_executed(tr);
92 `uvm_do_callbacks(apb_master,apb_master_cbs,trans_executed(this,tr))
93 
94 seq_item_port.item_done();
95 ->trig ;
96 end
97 endtask: run_phase

100 output logic [31:0] data);

103 this.sigs.mck.pwrite <= '0;

42 `uvm_component_utils(apb_monitor)

44 function new(string name, uvm_component parent = null);
45 super.new(name, parent);
46 ap = new("ap", this);
47 endfunction: new
48 
49 virtual function void build_phase(uvm_phase phase);
50 apb_agent agent;
51 if ($cast(agent, get_parent()) && agent != null) begin
52 sigs = agent.vif;
53 end
54 else begin
55 virtual apb_if tmp;
56 if (!uvm_config_db#(apb_vif)::get(this, "", "vif", tmp)) begin
57 `uvm_fatal("APB/MON/NOVIF", "No virtual interface specified for this monitor instance")
58 end
59 sigs = tmp;
60 end
61 endfunction
62 
63 virtual task run_phase(uvm_phase phase);
64 super.run_phase(phase);
65 forever begin
66 apb_rw tr;
67
68 // Wait for a SETUP cycle
69 do begin
70 @ (this.sigs.pck);
71 end
72 while (this.sigs.pck.psel !== 1'b1 ||
73 this.sigs.pck.penable !== 1'b0);
74 
75 tr = apb_rw::type_id::create("tr", this);
76
77 tr.kind = (this.sigs.pck.pwrite) ? apb_rw::WRITE : apb_rw::READ;
78 tr.addr = this.sigs.pck.paddr;
79 
80 @ (this.sigs.pck);
81 if (this.sigs.pck.penable !== 1'b1) begin
82 `uvm_error("APB", "APB protocol violation: SETUP cycle not followed by ENABLE cycle");
83 end
84 tr.data = (tr.kind == apb_rw::READ) ? this.sigs.pck.prdata :
85 this.sigs.pck.pwdata;
86 
87 trans_observed(tr);
88 `uvm_do_callbacks(apb_monitor,apb_monitor_cbs,trans_observed(this,tr))
89 
90 ap.write(tr);
91 end
92 endtask: run_phase
93 
94 virtual protected task trans_observed(apb_rw tr);
95 endtask
96 
97endclass: apb_monitor
