Blocking Put & Get Ports

Classic point-to-point transaction communication between a producer and consumer component using blocking uvm_blocking_put_port and uvm_blocking_get_port.

Key UVM APIs: uvm_blocking_put_port uvm_blocking_get_port uvm_tlm_fifo

23/*
24About: producer_consumer
25
26this test is the basic simple to illustrates how to create a producer and consumer which are completely independent and connecting them using a uvm_tlm_fifo channel
27
28
29Walk through the test:
30
31two thrads *producer* and *consumer* will use uvm_blocking_put port and uvm_blocking_get port, to transfer an integer from the producer to the consumer through fifo exports at the connection function of an environment class
32
33
34*/

38//----------------------------------------------------------------------
39// module top
40//----------------------------------------------------------------------
41module test;
42 
43 import uvm_pkg::*;
44
45 //----------------------------------------------------------------------
46 // class producer
47 //----------------------------------------------------------------------
48 class producer extends uvm_component;
49 
50 uvm_blocking_put_port#(int) put_port;
51
52 function new(string name, uvm_component p = null);
53 super.new(name,p);
54 put_port = new("put_port", this);
55
56 endfunction
57
58 task run_phase(uvm_phase phase);
59
60 int randval;
61
62 for(int i = 0; i < 10; i++)
63 begin
64 randval = $random % 100;
65 #10;
66 `uvm_info("producer", $sformatf("sending %4d", randval), UVM_MEDIUM)
67 put_port.put(randval);
68 end
69 endtask
70
71 endclass : producer
72
73 //----------------------------------------------------------------------
74 // class consumer
75 //----------------------------------------------------------------------
76 class consumer extends uvm_component;
77 
78 uvm_blocking_get_port#(int) get_port;
79
80 function new(string name, uvm_component p = null);
81 super.new(name,p);
82 get_port = new("get_port", this);
83 endfunction
84
85 task run_phase(uvm_phase phase);
86
87 int val;
88
89 forever
90 begin
91 get_port.get(val);
92 `uvm_info("consumer", $sformatf("receiving %4d", val), UVM_MEDIUM)
93 end
94
95 endtask
96
97 endclass : consumer
98
99 //----------------------------------------------------------------------
100 // class env
101 //----------------------------------------------------------------------
102 class env extends uvm_env;
103 producer p;
104 consumer c;
105 uvm_tlm_fifo #(int) f;
106
107 function new(string name = "env");
108 super.new(name);
109 p = new("producer", this);
110 c = new("consumer", this);
111 f = new("fifo", this);
112 $display("fifo put_export: %s", f.m_name);
113 endfunction
114
115 function void connect_phase(uvm_phase phase);
116 p.put_port.connect(f.put_export);
117 c.get_port.connect(f.get_export);
118 endfunction
119
120 task run_phase(uvm_phase phase);
121 phase.raise_objection(this);
122 #1000;
123 phase.drop_objection(this);
124 endtask
125
126 endclass
