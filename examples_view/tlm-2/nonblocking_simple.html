Non-Blocking 4-Phase Protocol

Exemplifies the full IEEE 1666 TLM-2 non-blocking 4-phase transaction protocol (BEGIN_REQ, END_REQ, BEGIN_RESP, END_RESP) modeling USB bulk transfers without simulation deadlocks.

Key UVM APIs: uvm_tlm_nb_initiator_socket uvm_tlm_nb_target_socket nb_transport_fw nb_transport_bw uvm_tlm_time

Simulation Flow Stepper

Interactive Execution Progression
1BEGIN_REQ (Initiator -> Target)
2END_REQ (Target -> Initiator)
3BEGIN_RESP (Target -> Initiator)
4END_RESP (Initiator -> Target)

1. BEGIN_REQ (Initiator -> Target)

Target: host (initiator)

The host initiator initiates the transfer by calling `nb_transport_fw(xfer, BEGIN_REQ)`. The device endpoint records the transfer and responds `UVM_TLM_ACCEPTED`.

Expected Log Output: UVM_INFO @ 0 ns: host [TLM2] nb_transport_fw(xfer, BEGIN_REQ)

Testbench Architecture Topology

initiator host
USB Host
target device
USB Device Endpoint

39 `uvm_info("USB/DEV/FWD", $sformatf("%s @%0d: %s",
40 ph.name(), delay.get_realtime(1ns),
41 xfer.convert2string()), UVM_LOW)
42 
43 // This device?
44 if (xfer.addr != 7'h7A) return UVM_TLM_COMPLETED;
45 
46 // Valid endpoint in the device?
47 if (xfer.endp != 4'h5) return UVM_TLM_COMPLETED;

49 case (xfer.kind)
50 
51 usb_xfer::OUT: begin
52 case (ph)
53 USB_TLM_TOKEN: begin
54 return UVM_TLM_ACCEPTED;
55 end
56 
57 USB_TLM_DATA: begin
58 data = xfer.data;
59 
60 // Could complete transfer early here
61 fork: out_ack
62 automatic usb_xfer xf = xfer;
63 automatic uvm_tlm_time dl = delay;
64 begin
65 usb_tlm_phase ph = USB_TLM_HANDSHAKE;
66 #100;
67 xf.status = usb_xfer::ACK;
68 assert(sock.nb_transport_bw(xf, ph, dl) ==
69 UVM_TLM_COMPLETED);
70 
71 `uvm_info("USB/DEV/OUT/DONE", xf.convert2string(),
72 UVM_NONE);
73
74 end
75 join_none
76 
77 return UVM_TLM_ACCEPTED;
78 end
79
80 endcase
81 end
82 
83 usb_xfer::IN: begin
84 case (ph)
85 USB_TLM_TOKEN: begin
86 // Could return the data early here
87 fork: in_data
88 automatic usb_xfer xf = xfer;
89 automatic uvm_tlm_time dl = delay;
90 begin
91 usb_tlm_phase ph = USB_TLM_DATA;
92 #150;
93 begin
94 byte tdata[2]='{'hAB, 'hCD};
95 xf.data = tdata;
96 end
97 void'(sock.nb_transport_bw(xf, ph, dl));
98 end
99 join_none

102 end

38 function uvm_tlm_sync_e nb_transport_bw(usb_xfer xfer,
39 ref usb_tlm_phase ph,
40 input uvm_tlm_time delay);
41 
42 `uvm_info("USB/HST/BWD", $sformatf("%s @%0d: %s",
43 ph.name(), delay.get_realtime(1ns),
44 xfer.convert2string()), UVM_LOW)
45
46 assert(xfer == this.xfer);
47 this.ph = ph;
48 this.delay = delay;

50 case (xfer.kind)
51 
52 usb_xfer::OUT: begin
53 assert(ph == USB_TLM_HANDSHAKE);
54 sync = UVM_TLM_COMPLETED;
55 return UVM_TLM_COMPLETED;
56 end

58 usb_xfer::IN: begin
59 assert(ph == USB_TLM_DATA);

61 return UVM_TLM_ACCEPTED;
62 end

68 //
69 // Execute an OUT followed by an IN bulk transfer
70 //
71 virtual task run_phase(uvm_phase phase);
72 
73 phase.raise_objection(this);
74
75 xfer = usb_xfer::type_id::create("xfer",,get_full_name());
76 
77 // OUT bulk transfer
78
79 do begin
80 xfer.kind = usb_xfer::OUT;
81 xfer.addr = 7'h7A;
82 xfer.endp = 4'h5;
83
84 ph = USB_TLM_TOKEN;
85 sync = sock.nb_transport_fw(xfer, ph, delay);
86 if (sync == UVM_TLM_COMPLETED) begin
87 `uvm_info("USB/HST/OUT/REFUSED", "Device refused the transfer",
88 UVM_LOW);
89 break;
90 end
91 
92 ph = USB_TLM_DATA;
93 begin
94 byte tdata[4]='{'hDE, 'hAD, 'hBE, 'hEF};
95 xfer.data = tdata;
96 end
97 sync = sock.nb_transport_fw(xfer, ph, delay);
98 if (sync == UVM_TLM_COMPLETED) begin
99 `uvm_info("USB/HST/OUT/EARLY", "Device completed OUT transfer early",
100 UVM_LOW);
101 break;
102 end
103 
104 // Wait for the device to reply
105 wait (sync == UVM_TLM_COMPLETED);
106 `uvm_info("USB/HST/OUT/NORMAL", "OUT Transfer completed normally",
107 UVM_LOW);
108
109 end while (0);
110 
111 `uvm_info("USB/HST/OUT/DONE", xfer.convert2string(), UVM_NONE);
112 
113 // IN bulk transfer
114
115 do begin
116 // Ok to reuse the same XFER instance
117 xfer.kind = usb_xfer::IN;
118 xfer.addr = 7'h7A;
119 xfer.endp = 4'h5;
120 xfer.data = new[0];
121
122 ph = USB_TLM_TOKEN;
123 sync = sock.nb_transport_fw(xfer, ph, delay);
124 if (sync == UVM_TLM_COMPLETED) begin
125 `uvm_info("USB/HST/IN/REFUSED", "Device refused the transfer",
126 UVM_LOW);
127 break;
128 end
129 if (sync == UVM_TLM_UPDATED) begin
130 `uvm_info("USB/HST/IN/EARLY", "Device returned bulk data early",
131 UVM_LOW);
132 end
133 else begin
134 // Wait for the device to reply
135 wait(ph == USB_TLM_DATA);
136 `uvm_info("USB/HST/IN/NORMAL", "IN data returned normally",
137 UVM_LOW);
138 end
139 
140 ph = USB_TLM_HANDSHAKE;
141 xfer.status = usb_xfer::ACK;
142 sync = sock.nb_transport_fw(xfer, ph, delay);
143 assert(sync == UVM_TLM_COMPLETED);
144
145 end while (0);
146 
147 `uvm_info("USB/HST/IN/DONE", xfer.convert2string(), UVM_NONE);
148 
149 phase.drop_objection(this);
150 endtask

2This simple nonblocking examples uses a USB 2.0 BULK transfer as an example.
3 
4Instead of translating the transfer into a series of packets transmitted
5back and forth between the host and device, a single transfer is exchanged
6and the timing points (i.e. calls to nb_transport_fw/bw()) are used in lieu
7packet transfers.
8 
9For a normal host->device bulk transfer, the following timing points occur
10until a UVM_TLM_COMPLETED sync value is received
11by either the host or the device.
12 
131. nb_transport_fw(xfer, ph)
14 - xfer.kind == OUT
15 - ph == USB_TLM_TOKEN
16 > Device responds with UVM_TLM_ACCEPTED, or
17 > Device responds with UVM_TLM_COMPLETED
18 
192. nb_transport_fw(xfer, ph)
20 - xfer.data contains bulk data
21 - ph == USB_TLM_DATA
22 > Device responds with UVM_TLM_ACCEPTED, or
23 > Device responds with UVM_TLM_COMPLETED, xfer.status == ACK/NAK/STALL/NYET
24 and ph updated to UVM_TLM_HANDSHAKE
25 
263. nb_transport_bw(xfer, ph)
27 - xfer.status == ACK/NACK/STALL/NYET
28 - ph == USB_TLM_HANDSKAHE
29 > Host responds with UVM_TLM_COMPLETED
30 
31For a device->host bulk transfer, the following timing points occur
32until a UVM_TLM_COMPLETED sync value is received
33by either the host or the device.
34 
351. nb_transport_fw(xfer, ph)
36 - xfer.kind == IN
37 - ph == USB_TLM_TOKEN
38 > Device responds with UVM_TLM_ACCEPTED, or
39 > Device responds with UVM_TLM_UPDATED, xfer.data contains bulk data
40 and ph updated to UVM_TLM_DATA (goto step 3), or
41 > Device responds with UVM_TLM_COMPLETED, xfer.status == NAK/STALL
42 and ph updated to UVM_TLM_HANDSHAKE
43 
442. nb_transport_bw(xfer, ph)
45 - xfer.data contains bulk data
46 - ph == USB_TLM_DATA
47 > Host responds with UVM_TLM_ACCEPTED, or
48 > Host responds with UVM_TLM_COMPLETED, xfer.status == ACK
49 and ph updated to UVM_TLM_HANDSHAKE
50 
513. nb_transport_fw(xfer, ph)
52 - xfer.status == ACK
53 - ph == USB_TLM_HANDSKAHE
54 > Device responds with UVM_TLM_COMPLETED
