Module-Based UVM

Illustrates executing UVM testbench hierarchies directly from pure SystemVerilog modules and calling run_test without a formal top-level test class.

Key UVM APIs: uvm_component run_test uvm_pkg

23import uvm_pkg::*;
24`include "uvm_macros.svh"

26//----------------------------------------------------------------------
27// lower
28//----------------------------------------------------------------------
29class lower extends uvm_component;
30 int data;
31 string str;
32 
33 function new (string name, uvm_component parent);
34 super.new(name, parent);
35 endfunction
36 
37 task run_phase(uvm_phase phase);
38 phase.raise_objection(this);
39 #10 $display("%0t: %s HI", $time, get_full_name());
40 phase.drop_objection(this);
41 endtask
42 
43 function string get_type_name();
44 return "lower";
45 endfunction
46 
47 function void build_phase(uvm_phase phase);
48 void'(uvm_config_int::get(this, "", "data", data));
49 void'(uvm_config_string::get(this, "", "str", str));
50 endfunction
51 
52 function void do_print(uvm_printer printer);
53 printer.print_field("data", data, 32);
54 printer.print_string("str", str);
55 endfunction
56endclass

58//----------------------------------------------------------------------
59// myunit
60//----------------------------------------------------------------------
61class myunit extends uvm_component;
62 lower l1;
63 lower l2;
64 int a[];
65 
66 function new (string name, uvm_component parent);
67 super.new(name, parent);
68 uvm_config_string::set(this, "l1", "str", "hi");
69 uvm_config_int::set(this, "*", "data", 'h100);
70 l1 = new ("l1", this);
71 l2 = new ("l2", this);
72 l1.data = 'h30;
73 l2.data = 'h40;
74 a = new[5]; for(int i=0; i<5;++i) a[i] = i*i;
75 endfunction
76 
77 task run_phase(uvm_phase phase);
78 //Check config from initial block
79 if(l1.data != 55)
80 `uvm_error("BADCFG", $sformatf("Expected l1.data = 55, got %0d", l1.data))
81 if(l2.data != 101)
82 `uvm_error("BADCFG", $sformatf("Expected l2.data = 101, got %0d", l2.data))
83 if(l1.str != l2.str && l1.str != "hi")
84 `uvm_error("BADCFG", $sformatf("Expected l1.str = \"hi\" and l2.str = \"hi\", got l1.str = \"%s\" and l2.str = \"%s\"", l1.str, l2.str))
85 phase.raise_objection(this);
86 #10 $display("%0t: %s HI", $time, get_full_name());
87 phase.drop_objection(this);
88 endtask
89 
90 function string get_type_name();
91 return "myunit";
92 endfunction
93 
94 function void do_print(uvm_printer printer);
95 printer.print_array_header("a", a.size());
96 for(int i=0; i<a.size(); ++i)
97 printer.print_field( $sformatf("a[%0d]", i), a[i], 32, UVM_HEX, "[");
98 printer.print_array_footer();
99 endfunction
100
101endclass

128module top;
129 import uvm_pkg::*;
130 import user_pkg::*;
131 
132 myunit mu = new("mu", null);
133 mydata bar = new;
134 
135 initial begin
136 uvm_top.finish_on_completion = 0;
137 uvm_config_int::set(null, "mu.*", "data", 101);
138 uvm_config_string::set(null, "mu.*", "str", "hi");
139 uvm_config_int::set(null, "mu.l1", "data", 55);
140 uvm_config_object::set(null, "mu.*", "obj", bar);
141 mu.print();
142 run_test();
143 mu.print();
144 end
145endmodule
